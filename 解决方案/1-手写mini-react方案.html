手写 Mini-React(理解原理) ​

通过 200 行实现,理解 React 的核心机制。

一、最小实现 ​

完整代码:mini-react/index.js

js
// ============================================
// 核心实现:约 200 行
// ============================================

// 1. createElement:JSX 编译后调用
function createElement(type, props, ...children) {
  return {
    type,
    props: {
      ...props,
      children: children.map(c =>
        typeof c === 'object' ? c : createTextElement(c)
      ),
    },
  }
}

function createTextElement(text) {
  return {
    type: 'TEXT_ELEMENT',
    props: {
      nodeValue: text,
      children: [],
    },
  }
}

// 2. createDom:从 VNode 创建真实 DOM
function createDom(vnode) {
  const dom = vnode.type === 'TEXT_ELEMENT'
    ? document.createTextNode('')
    : document.createElement(vnode.type)

  updateDom(dom, {}, vnode.props)

  // 递归创建子节点
  vnode.props.children.forEach(child =>
    dom.appendChild(createDom(child))
  )

  return dom
}

function updateDom(dom, prevProps, nextProps) {
  // 删除旧属性
  Object.keys(prevProps)
    .filter(key => key !== 'children')
    .forEach(key => delete dom[key])

  // 添加/更新新属性
  Object.keys(nextProps)
    .filter(key => key !== 'children')
    .forEach(key => {
      if (key.startsWith('on')) {
        const eventType = key.toLowerCase().substring(2)
        dom.addEventListener(eventType, nextProps[key])
      } else {
        dom[key] = nextProps[key]
      }
    })
}

// 3. 调和(Reconciliation):对比新旧 VNode
function reconcile(parentDom, oldVNode, newVNode, index = 0) {
  if (!oldVNode && !newVNode) return

  // 删除
  if (!newVNode) {
    parentDom.removeChild(parentDom.childNodes[index])
    return
  }

  // 新增
  if (!oldVNode) {
    parentDom.appendChild(createDom(newVNode))
    return
  }

  // 类型变化 → 整体替换
  if (oldVNode.type !== newVNode.type) {
    parentDom.replaceChild(createDom(newVNode), parentDom.childNodes[index])
    return
  }

  // 同类型 → 更新属性 + 递归子节点
  const sameDom = parentDom.childNodes[index]
  updateDom(sameDom, oldVNode.props, newVNode.props)
  newVNode.props.children.forEach((newChild, i) =>
    reconcile(sameDom, oldVNode.props.children[i], newChild, i)
  )
}

// 4. 渲染入口
function render(vnode, container) {
  // 保存上一次的 VNode
  const prevVNode = container._vnode
  container._vnode = vnode

  if (!prevVNode) {
    // 首次渲染
    container.appendChild(createDom(vnode))
  } else {
    // 更新
    reconcile(container, prevVNode, vnode)
  }
}

// 5. 简易 state
let stateHook = []
let hookIndex = 0

function useState(initial) {
  const idx = hookIndex
  stateHook[idx] = stateHook[idx] ?? initial

  const setState = (newState) => {
    stateHook[idx] = newState
    // 触发 re-render
    render(currentRoot._vnode, document.getElementById('root'))
  }

  hookIndex = 0  // reset for next render
  return [stateHook[idx], setState]
}

// 6. 应用入口
let currentRoot = null

function App() {
  const [count, setCount] = useState(0)

  return createElement('div', null,
    createElement('h1', null, 'Count: ', count),
    createElement('button', { onClick: () => setCount(count + 1) }, '+1')
  )
}

const container = document.getElementById('root')
currentRoot = container
render(createElement(App, null), container)

二、核心原理解析 ​

2.1 createElement 做了什么 ​

jsx
// JSX
<div className="box">Hello</div>

// 编译后
createElement('div', { className: 'box' }, 'Hello')

// 返回
{
  type: 'div',
  props: {
    className: 'box',
    children: [{ type: 'TEXT_ELEMENT', props: { nodeValue: 'Hello', children: [] } }],
  },
}

JSX 是 createElement 的语法糖。

2.2 reconcile 三种情况 ​

js
// 1. 新增(没有老节点)
if (!oldVNode) {
  parentDom.appendChild(createDom(newVNode))
  return
}

// 2. 删除(没有新节点)
if (!newVNode) {
  parentDom.removeChild(parentDom.childNodes[index])
  return
}

// 3. 类型变化 → 替换
if (oldVNode.type !== newVNode.type) {
  parentDom.replaceChild(createDom(newVNode), parentDom.childNodes[index])
  return
}

// 4. 同类型 → 更新
updateDom(dom, oldVNode.props, newVNode.props)
reconcileChildren(dom, oldChildren, newChildren)

2.3 调和算法的关键 ​

- 同级比较,跨层级不递归
- 类型变化整体替换
- 同类型复用,只更新属性和子节点
- 子节点按 index 匹配(简化版,真实用 key)

三、扩展:加 key 支持 ​

js
function reconcileChildren(parentDom, oldChildren, newChildren) {
  // 用 key 匹配,而不是 index
  const maxLength = Math.max(oldChildren.length, newChildren.length)

  for (let i = 0; i < maxLength; i++) {
    reconcile(parentDom, oldChildren[i], newChildren[i], i)
  }
}

// 实际 React 会:
// 1. 遍历新 children,尝试匹配老的 key
// 2. 复用匹配上的,移动位置
// 3. 删除没匹配的,新建缺失的

四、扩展:加 fiber ​

js
// 每个 fiber 有指向下一个工作单元的指针
function performUnitOfWork(fiber) {
  // 1. 处理当前 fiber
  // 2. 返回下一个 fiber(子 或 sibling 或 uncle)
}

// workLoop 会被分成多个时间片
function workLoop(deadline) {
  let shouldYield = false
  while (nextUnitOfWork && !shouldYield) {
    nextUnitOfWork = performUnitOfWork(nextUnitOfWork)
    shouldYield = deadline.timeRemaining() < 1
  }
  if (!nextUnitOfWork && workInProgress) {
    commitRoot()
  }
  requestIdleCallback(workLoop)
}

五、扩展:加 useState ​

js
let wipFiber = null
let hookIndex = 0
const hooks = []

function useState(initial) {
  const oldHook = wipFiber?.alternate?.hooks?.[hookIndex]

  const hook = {
    state: oldHook ? oldHook.state : initial,
    queue: [],
  }

  // 处理 actions
  hook.queue.forEach(action => {
    hook.state = typeof action === 'function' ? action(hook.state) : action
  })
  hook.queue = []

  const setState = (action) => {
    hook.queue.push(action)
    // 触发 re-render
    workInProgress = {
      type: wipFiber.type,
      stateNode: wipFiber.stateNode,
      props: wipFiber.props,
      alternate: wipFiber,
      hooks: [],
    }
    nextUnitOfWork = workInProgress
    hookIndex = 0
  }

  wipFiber.hooks.push(hook)
  hookIndex++
  return [hook.state, setState]
}

六、真实 React 比 Mini 版多了什么 ​

1. Concurrent Mode(时间切片、优先级)
2. Suspense(异步组件)
3. Hooks 完整实现(useEffect、useMemo 等)
4. Server Components(服务端组件)
5. TypeScript 类型推导
6. DevTools 支持
7. 错误边界(Error Boundary)
8. Fragment、Portal、Context
9. 完整的 Diff 算法(包含 key 复用)
10. Concurrent Reconciliation

七、面试价值 ​

Q1:React 是怎么工作的? ​

答:

  • 组件 render() 返回 VNode(JS 对象)
  • React 用 diff 算法对比新旧 VNode
  • 把差异(patches)应用到真实 DOM
  • 关键优化:同层比较、类型判断、key 匹配

Q2:fiber 解决了什么问题? ​

答:

  • 老的 Stack Reconciler 不可中断,长任务卡 UI
  • Fiber 把渲染拆成多个单元,可让出主线程
  • 实现 Concurrent Mode:用户优先、并发渲染

Q3:为什么 React 不直接操作 DOM? ​

答:

  • DOM 操作慢,批量更新更快
  • Virtual DOM 提供跨平台能力(React Native)
  • 状态管理更清晰(数据驱动)
  • 但有性能开销(简单场景不如 Svelte/Solid.js)

面试追问(深化原 Q1-Q3) ​

  • mini-react 拆几个模块? render(VDOM→DOM)/ commit(patch)/ reconcile(diff)/ event(委托)/ hooks(链表)
  • createElement 返回什么? { type, props: { children } };type 是字符串(原生)或函数(组件)
  • mini-diff 难点? 同级 key 匹配;类型不同整体替换;同类型只对比 props
  • 实现 useState 关键? 链表存 hook;setState push updateQueue,触发 re-render
  • 实现 useEffect 关键? commit 后异步执行;依赖浅比对;cleanup 存到 hook

Q4: 实现 useState 要点? ​

答:

  • 链表存 hooks;按调用顺序索引
  • 首次调用:创建 hook 存 state;后续:读 state
  • setState:push 到 updateQueue,触发 re-render
  • 难点:闭包内 setState 怎么拿到最新 state → 用 updateQueue 队列累加

Q5: 实现 useEffect 要点? ​

答:

  • mount/update 时把 effect 存到 fiber.memoizedState
  • commit 后异步执行(用 setTimeout / scheduler.postTask 模拟)
  • 依赖数组浅比对 → 不等就重新执行
  • cleanup 函数在下一次执行前调用(组件卸载也调用)

🎯 面试常考 ​

Q1：手写 mini-React 的核心难点在哪? ​

三个核心:

  1. Fiber 结构:把递归的 Reconciler 改成可中断的链表遍历 —— 每个 fiber 节点记录 return/child/sibling,靠 sibling 串成链表
  2. 双缓冲 + 提交阶段:render 阶段在 workInProgress 上算,commit 阶段一次性同步到 DOM —— 这是「可中断但最终一致」的来源
  3. Hooks 的实现:用链表存每个组件的 hook,靠 hook.next 串起来, 每次 render 按顺序遍历,天然对应「Hooks 不能写在条件里」的限制
ts
type Fiber = {
  type: any; stateNode: any; props: any
  return: Fiber | null
  child: Fiber | null
  sibling: Fiber | null
  hooks?: Hook[]
  flags: number          // 副作用标记
}

为什么用链表而不是树:链表可以从任意节点暂停(记住 current 指针), 树不行。

Q2：Hooks 为什么不能写在 if 里? ​

因为 React 靠调用顺序匹配 hook,不是靠名字。

每次 render 时,React 用一个全局指针 currentHook 顺序遍历 fiber 的 hook 链表, 把第 N 次调用 useState 和链表第 N 个节点配对。

ts
function Counter() {
  const [a, setA] = useState(0)     // 链表节点 1
  if (props.cond) {
    const [b, setB] = useState(0)   // ❌ 这次可能没执行,链表就错位了
  }
  const [c, setC] = useState(0)     // 这里的 c 会匹配到上次的 b
}

后果:状态串位、useEffect 数量对不上、报错「Rendered fewer hooks than expected」。

规避:无条件调用所有 hook,条件逻辑放进 hook 内部。

📝 小结 ​

  • 手写 React 的三个核心:Fiber 链表结构、可中断的双缓冲 + commit 阶段、Hooks 的链表实现。
  • 用链表而非树,是因为链表能从任意节点暂停恢复,这是可中断渲染的基础。
  • Hooks 不能写在条件里,是因为 React 靠调用顺序匹配而不是靠变量名。
  • render 阶段算 workInProgress,commit 阶段一次性提交,所以「可中断」但「最终一致」。
