2.2 查询参数 ​

tsx
import { useSearchParams } from 'react-router-dom'

function Search() {
  const [params, setParams] = useSearchParams()
  const q = params.get('q')  // /search?q=vue
  
  function updateQuery(key: string, value: string) {
    setParams({ [key]: value })
  }
  
  return <input value={q ?? ''} onChange={e => updateQuery('q', e.target.value)} />
}

五、路由守卫(Protected Routes) ​

5.1 鉴权路由 ​

tsx
import { Navigate, useLocation } from 'react-router-dom'

function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const user = useUser()
  const location = useLocation()

  if (!user.isLoggedIn) {
    return <Navigate to="/login" state={{ from: location }} replace />
  }

  return <>{children}</>
}

// 使用
<Route path="/dashboard" element={
  <ProtectedRoute>
    <Dashboard />
  </ProtectedRoute>
} />

5.2 角色权限 ​

tsx
function RoleRoute({ roles, children }) {
  const user = useUser()
  if (!user || !roles.includes(user.role)) {
    return <Navigate to="/403" replace />
  }
  return <>{children}</>
}

// 使用
<Route path="/admin" element={
  <RoleRoute roles={['admin', 'super_admin']}>
    <AdminPanel />
  </RoleRoute>
} />

5.3 懒加载 + Suspense ​

tsx
import { lazy, Suspense } from 'react'

const Dashboard = lazy(() => import('./pages/Dashboard'))
const Admin = lazy(() => import('./pages/Admin'))

<Suspense fallback={<Loading />}>

    <Route path="/dashboard" element={<Dashboard />} />
    <Route path="/admin" element={<Admin />} />
  </Routes>
</Suspense>

import { useLocation } from 'react-router-dom'

function Component() {
  const location = useLocation()
  
  location.pathname    // '/users/123'
  location.search      // '?tab=profile'

  location.state       // 导航时传的 state
  location.key         // 唯一 key,用于判断路由变化
}

八、实战模式 ​

8.1 鉴权封装 ​

tsx
// hooks/useAuthGuard.ts
import { useEffect } from 'react'
import { useNavigate, useLocation } from 'react-router-dom'
import { useUser } from './useUser'

export function useAuthGuard() {
  const user = useUser()
  const navigate = useNavigate()
  const location = useLocation()

  useEffect(() => {
    if (!user.isLoggedIn && !isPublicPath(location.pathname)) {
      navigate('/login', {
        replace: true,
        state: { from: location.pathname },
      })
    }
  }, [user.isLoggedIn, location.pathname])
}

十、面试常考 ​

Q1:BrowserRouter vs HashRouter? ​

答:

  • BrowserRouter:/about,URL 干净,需服务端 fallback
  • HashRouter:/#/about,无需配置,SEO 略差
  • 新项目用 BrowserRouter

Q1:如何做路由鉴权? ​

答:

  • 包裹 ProtectedRoute 组件
  • 未登录 → Navigate to /login
  • 把原路径放在 state,登录后跳回

Q3:lazy + Suspense 用法? ​

答:

tsx
const Page = lazy(() => import('./Page'))

<Suspense fallback={<Loading />}>
  <Page />
</Suspense>

面试追问(深化原 Q1-Q3) ​

  • BrowserRouter 服务端 fallback 怎么配? Nginx 配 try_files $uri /index.html;或用 Vercel/Netlify 默认支持
  • HashRouter 优点? 不需要服务端配置;旧项目无服务端权限时可用
  • ProtectedRoute 怎么拿原路径? <Navigate to="/login" state="{​{ from: location }​}" />;登录后从 location.state.from 跳回
  • lazy + Suspense 怎么配合 SSR? 需要 loadable 或框架级方案(Next.js dynamic import),避免服务端拿不到 chunk
  • 路由懒加载的边界? 路由级 chunk 粒度适合;过细(如每个组件)反而增加请求数

Q4: 路由懒加载 vs 组件懒加载? ​

答:

  • 路由级:React.lazy(() => import('./page')) + Suspense;按页面分包,首屏快
  • 组件级:React.lazy 包裹大组件(如富文本、图表);用时才加载
  • 不要无限拆:每个 chunk < 50KB 才划算(网络往返成本)
