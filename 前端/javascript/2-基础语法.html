1.2 类型判断 ​

ts
// ❌ typeof 对 array / null / Date 全部返回 'object'，不够用
typeof []  // 'object'
typeof new Date()  // 'object'

// ✅ 最佳方案：Object.prototype.toString.call（最准确）
function getType(value: unknown): string {
  return Object.prototype.toString.call(value).slice(8, -1).toLowerCase()
}
getType([])             // 'array'
getType(new Date())     // 'date'
getType(/regex/)        // 'regex'
getType(new Map())      // 'map'
getType(null)           // 'null'
getType(undefined)      // 'undefined'

// ✅ instanceof：检测原型链
[] instanceof Array           // true
new Date() instanceof Date    // true

// ✅ Array.isArray：数组专用
Array.isArray([])  // true

2. 类型转换（高频坑） ​

2.1 隐式转换规则 ​

js
// 一元 + 触发 Number() 转换

+''        // 0    ← 注意：空字符串变 0
+'abc'     // NaN

// == 的隐式转换规则（面试经典题）

3. 作用域与闭包 ​

3.1 词法作用域 ​

JS 用词法作用域（写代码时确定，不是运行时）。

js
const name = 'global'

function outer() {
  const name = 'outer'
  function inner() {
    // 关键：inner 看到的 name 是它定义时的 outer，不是调用时的 global
    console.log(name)  // 'outer'
  }
  inner()
}

outer()

3.2 闭包 ​

闭包 = 函数 + 其词法环境。函数能"记住"定义时的变量。

js
function counter() {
  let count = 0
  return function () {
    count++
    return count
  }
}

const c = counter()
c()  // 1
c()  // 2
c()  // 3
// count 不会被回收，因为内部函数还在引用

闭包实际应用：

js
// 1. 数据私有化（模拟私有变量）
function createWallet(initial: number) {
  let balance = initial
  return {
    deposit(amount: number) { balance += amount },
    withdraw(amount: number) {
      if (amount > balance) throw new Error('余额不足')
      balance -= amount
    },
    getBalance() { return balance },
  }
}

// 2. 函数柯里化
function curry(fn: Function) {
  return function curried(...args: any[]) {
    if (args.length >= fn.length) return fn.apply(this, args)
    return (...rest: any[]) => curried.apply(this, [...args, ...rest])
  }
}

const add = (a: number, b: number, c: number) => a + b + c
const curriedAdd = curry(add)
curriedAdd(1)(2)(3)  // 6

// 3. 防抖节流（高频）
function debounce(fn: Function, delay: number) {
  let timer: number | null = null
  return function (...args: any[]) {
    if (timer) clearTimeout(timer)
    timer = setTimeout(() => fn.apply(this, args), delay) as unknown as number
  }
}

4. this 指向（4 种规则） ​

js
// 1. 默认绑定：独立函数调用，this 指向 window（非严格模式）/ undefined（严格模式）
function fn() { console.log(this) }
fn()  // window 或 undefined

// 2. 隐式绑定：谁调用，this 指向谁
const obj = {
  name: 'tom',
  greet() { console.log(this.name) },  // 'tom'
}
obj.greet()

// 3. 显式绑定：call / apply / bind
function greet() { console.log(this.name) }
const obj2 = { name: 'jerry' }
greet.call(obj2)      // 'jerry'
greet.apply(obj2)     // 'jerry'
const bound = greet.bind(obj2)
bound()                // 'jerry'

// 4. new 绑定：this 指向新创建的对象
function Person(name: string) {
  this.name = name
}
const p = new Person('tom')
console.log(p.name)  // 'tom'

优先级：new > 显式 > 隐式 > 默认

js
// 经典坑：定时器里的 this

  name: 'tom',

5. 原型与原型链 ​

5.1 核心概念 ​

js
// 每个对象都有 __proto__，指向其构造函数的 prototype
const obj = {}
obj.__proto__ === Object.prototype  // true

// 每个函数都有 prototype（箭头函数除外）
function Foo() {}
const f = new Foo()
f.__proto__ === Foo.prototype  // true

// 原型链终点：null
Object.prototype.__proto__  // null

5.2 经典图 ​

f (实例) → Foo.prototype → Object.prototype → null

5.3 实际应用 ​

js
// 1. 继承：组合继承（call + prototype）
function Animal(name: string) {
  this.name = name
}
Animal.prototype.say = function () {
  console.log(`I am ${this.name}`)
}

function Dog(name: string, breed: string) {
  Animal.call(this, name)  // 继承属性
  this.breed = breed
}
Dog.prototype = Object.create(Animal.prototype)  // 继承方法
Dog.prototype.constructor = Dog  // 修复 constructor

// 2. class 语法糖（ES6，推荐）
class Animal {
  constructor(name: string) { this.name = name }
  say() { console.log(`I am ${this.name}`) }
}

class Dog extends Animal {
  constructor(name: string, public breed: string) {
    super(name)
  }
  bark() { console.log('Woof!') }
}

7. 深拷贝 vs 浅拷贝 ​

ts
// 浅拷贝：只复制一层
const obj = { a: 1, b: { c: 2 } }
const shallow = { ...obj }
shallow.b.c = 3
obj.b.c  // 3 ← 原对象也被改了

// 深拷贝方案
// 1. JSON（简单但有缺陷：不能拷贝函数、Date、RegExp、循环引用）
const deep1 = JSON.parse(JSON.stringify(obj))

// 2. structuredClone（现代浏览器原生支持，2022+）
const deep2 = structuredClone(obj)

// 3. 自实现（兼容老环境）
function deepClone<T>(value: T, hash = new WeakMap()): T {
  if (value === null || typeof value !== 'object') return value
  if (value instanceof Date) return new Date(value) as any
  if (value instanceof RegExp) return new RegExp(value) as any
  if (hash.has(value as any)) return hash.get(value as any)

  const isArray = Array.isArray(value)
  const cloneObj = isArray ? [] : Object.create(Object.getPrototypeOf(value))
  hash.set(value as any, cloneObj)

  for (const key of Reflect.ownKeys(value as any)) {
    (cloneObj as any)[key] = deepClone((value as any)[key], hash)
  }
  return cloneObj as T
}

面试追问 ​

  • typeof 能识别哪些类型? undefined boolean number string bigint symbol function object;null 识别为 object
  • 闭包内存泄漏? 闭包不释放引用链上的变量;手动置 null 解除;现代 V8 优化
  • 原型链继承的 3 种方式? 构造函数继承 / 原型链继承 / 组合继承;ES6 class 是语法糖
  • this 优先级? new 绑定 > 显式(call/apply/bind) > 隐式(obj.fn()) > 默认(严格模式 undefined)
  • structuredClone 限制? 不能克隆函数、Symbol、DOM 节点、循环引用已修复

真实项目踩过的坑 ​

  1. forEach 不等 await：批量上传、批量校验要改用 Promise.all(arr.map(...)) 或 for...of。
  2. JSON.parse(JSON.stringify(obj)) 丢数据：会丢失 undefined、函数、Symbol、Date 会变字符串。
  3. new Date().getMonth() 返回 0-11：1 月是 0，不是 1。这是 JS 历史 bug。
  4. 数组 sort 默认按字符串排序：[10, 2, 1].sort() → [1, 10, 2]，数字必须传比较函数。
  5. parseInt 不传第二参数会出问题：parseInt('08') 老浏览器会按八进制解析，必须写 parseInt('08', 10)。
