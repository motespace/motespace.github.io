3.4 picker（选择器） ​

vue
<template>
  <view>
    <!-- 普通选择器 -->
    <picker
      mode="selector"
      :range="array"
      :value="index"
      @change="onChange"
    >
      <view>当前选择: {{ array[index] }}</view>
    </picker>

    <!-- 多列选择器 -->
    <picker
      mode="multiSelector"
      :range="multiArray"
      :value="multiIndex"
      @change="onMultiChange"
    >
      <view>多列选择</view>
    </picker>

    <!-- 时间选择器 -->
    <picker
      mode="time"
      :value="time"
      start="09:00"
      end="18:00"
      @change="onTimeChange"
    >
      <view>选择时间: {{ time }}</view>
    </picker>

    <!-- 日期选择器 -->
    <picker
      mode="date"
      :value="date"
      start="2020-01-01"
      end="2030-12-31"
      @change="onDateChange"
    >
      <view>选择日期: {{ date }}</view>
    </picker>

    <!-- 省市区选择器 -->
    <picker
      mode="region"
      :value="region"
      @change="onRegionChange"
    >
      <view>选择地区: {{ region[0] }} {{ region[1] }} {{ region[2] }}</view>
    </picker>
  </view>
</template>

<script setup>
import { ref } from 'vue'

const array = ['JavaScript', 'TypeScript', 'Vue', 'React']
const index = ref(0)

const multiArray = [
  ['上衣', '裤子', '鞋子'],
  ['S', 'M', 'L', 'XL'],
  ['红色', '蓝色', '黑色']
]
const multiIndex = ref([0, 0, 0])

const time = ref('12:00')
const date = ref('2024-01-01')
const region = ref(['广东省', '深圳市', '南山区'])

const onChange = (e) => {
  index.value = e.detail.value
  console.log('选择:', array[e.detail.value])
}

const onMultiChange = (e) => {
  multiIndex.value = e.detail.value
}

const onTimeChange = (e) => {
  time.value = e.detail.value
}

const onDateChange = (e) => {
  date.value = e.detail.value
}

const onRegionChange = (e) => {
  region.value = e.detail.value
}
</script>

四、媒体组件 ​

4.1 image（图片） ​

vue
<template>
  <view>
    <!-- 基础用法 -->
    <image src="/static/logo.png" mode="aspectFit" />

    <!-- mode 取值 -->
    <image src="/static/logo.png" mode="scaleToFill" /> <!-- 拉伸 -->
    <image src="/static/logo.png" mode="aspectFit" />   <!-- 适应 -->
    <image src="/static/logo.png" mode="aspectFill" />  <!-- 填充 -->
    <image src="/static/logo.png" mode="widthFix" />    <!-- 宽度固定 -->
    <image src="/static/logo.png" mode="heightFix" />    <!-- 高度固定 -->

    <!-- 懒加载 -->
    <image src="/static/logo.png" lazy-load />

    <!-- 显示默认图 -->
    <image
      src="/static/logo.png"
      :show-menu-by-longpress="true"
      @error="onImageError"
      @load="onImageLoad"
    />
  </view>
</template>

<script setup>
const onImageError = (e) => {
  console.error('图片加载失败', e)
}

const onImageLoad = (e) => {
  console.log('图片加载成功', e.detail.width, e.detail.height)
}
</script>

<style>
image {
  width: 300rpx;
  height: 300rpx;
}
</style>

mode 取值说明：

mode说明
scaleToFill缩放，不保持比例
aspectFit缩放，保持比例，最大完整显示
aspectFill缩放，保持比例，最大填充
widthFix宽度不变，高度自动变化
heightFix高度不变，宽度自动变化
top不缩放，只显示顶部
bottom不缩放，只显示底部
center不缩放，只显示中间
left不缩放，只显示左边
right不缩放，只显示右边
