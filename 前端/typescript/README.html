一、基础类型 ​

详细文档：数据类型、接口

类型说明示例
基础类型string, number, booleanlet name: string = 'test'
数组T[] 或 Array<T>let arr: number[] = [1, 2]
元组固定长度固定类型let t: [string, number]
枚举命名常量集合enum Color { Red, Green }
any任意类型let x: any
void无返回值function fn(): void
never从不返回function err(): never

接口核心概念 ​

  • 可选属性：age?: number
  • 只读属性：readonly id: number
  • 索引签名：[propName: string]: any
  • 函数类型：(x: number) => string
  • 继承：interface A extends B

二、泛型 ​

详细文档：泛型

核心概念 ​

tsx
// 泛型函数
function returnItem<T>(para: T): T {
  return para
}

// 泛型接口
interface ReturnItemFn<T> {
  (para: T): T
}

// 泛型类
class Stack<T> {
  private items: T[] = []
  push(item: T) { this.items.push(item) }
  pop(): T { return this.items.pop() }
}

泛型约束 ​

tsx
// 约束只能使用已有属性
function getValue<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key]
}

// 多类型约束
interface A { a: string }
interface B { b: number }
function fn<T extends A & B>(arg: T) { }

常用模式 ​

tsx
// 泛型默认值
interface Data<T = string> { value: T }

// 泛型别名
type Pair<T, U> = { first: T; second: U }

// 多泛型参数
function swap<T, U>(tuple: [T, U]): [U, T] {
  return [tuple[1], tuple[0]]
}

三、高级类型 ​

详细文档：[高级类型](./high type.md)

3.1 交叉类型 T & U ​

tsx
// 合并多个类型
interface A { a: string }
interface B { b: number }
type C = A & B // { a: string; b: number }

3.2 联合类型 T | U ​

tsx
// 多种类型之一
let value: string | number
value = 'hello' // ok
value = 123     // ok

3.3 类型别名 ​

tsx
type Name = string
type Callback = (data: string) => void
type Partial<T> = { [P in keyof T]?: T[P] }

3.4 条件类型 ​

tsx
T extends U ? X : Y

// 示例
type NonNullable<T> = T extends null | undefined ? never : T

// 分布式条件类型
type Extract<T, U> = T extends U ? T : never

3.5 映射类型 ​

tsx
// 内置工具类型
type Readonly<T> = { readonly [P in keyof T]: T[P] }
type Partial<T> = { [P in keyof T]?: T[P] }
type Required<T> = { [P in keyof T]-?: T[P] }
type Pick<T, K extends keyof T> = { [P in K]: T[P] }
type Omit<T, K extends keyof T> = Pick<T, Exclude<keyof T, K>>

3.6 类型推断 infer ​

tsx
// 推断返回值类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never

// 推断数组元素类型
type ElementType<T> = T extends (infer E)[] ? E : never

// 推断 Promise resolve 类型
type Awaited<T> = T extends Promise<infer U> ? U : T

3.7 模板字面量类型 ​

tsx
type EventName = 'click' | 'scroll'
type Handler = `on${EventName}` // 'onclick' | 'onscroll'

type PropEventSource<T> = {
  on<K extends string & keyof T>(eventName: `${K}`, handler: (value: T[K]) => void): void
}

四、Utility Types（工具类型） ​

详细文档：[高级类型](./high type.md)

4.1 内置工具类型 ​

类型说明示例
Partial<T>所有属性变为可选Partial<User>
Required<T>所有属性变为必需Required<Options>
Readonly<T>所有属性变为只读Readonly<Config>
Pick<T, K>选取部分属性Pick<User, 'id' | 'name'>
Omit<T, K>排除部分属性Omit<User, 'password'>
Record<K, T>创建对象类型Record<string, number>
Exclude<T, U>排除类型Exclude<'a' | 'b', 'a'>
Extract<T, U>提取类型Extract<'a' | 'b', 'a'>
NonNullable<T>排除 null/undefinedNonNullable<string | null>
ReturnType<T>函数返回值类型ReturnType<typeof fn>
Parameters<T>函数参数类型Parameters<typeof fn>
InstanceType<T>构造函数实例类型InstanceType<typeof Person>

4.2 自定义工具类型 ​

tsx
// 可控字段
type Mutable<T> = {
  -readonly [P in keyof T]: T[P]
}

// 解包 Promise
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T

// 获取对象深度属性
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P]
}

// 安全的 Promise 返回类型
type AsyncReturnType<T> = T extends (...args: any) => Promise<infer R> ? R : never

五、类型守卫 ​

详细文档：函数

5.1 typeof 类型守卫 ​

tsx
function padLeft(value: string | number) {
  if (typeof value === 'string') {
    return ' '.repeat(10) + value
  }
  return value.toFixed(2)
}

5.2 instanceof 类型守卫 ​

tsx
class Bird {
  fly() { console.log('fly') }
}
class Fish {
  swim() { console.log('swim') }
}

function move(pet: Bird | Fish) {
  if (pet instanceof Bird) {
    pet.fly()
  } else {
    pet.swim()
  }
}

5.3 自定义类型谓词 ​

tsx
interface Cat {
  meow(): void
}
interface Dog {
  bark(): void
}

function isCat(pet: Cat | Dog): pet is Cat {
  return (pet as Cat).meow !== undefined
}

function makeSound(pet: Cat | Dog) {
  if (isCat(pet)) {
    pet.meow()
  }
}

5.4 in 操作符 ​

tsx
interface A {
  a(): void
}
interface B {
  b(): void
}

function fn(x: A | B) {
  if ('a' in x) {
    x.a()
  } else {
    x.b()
  }
}

5.5 可辨识联合类型 ​

tsx
interface Square {
  kind: 'square'
  size: number
}
interface Circle {
  kind: 'circle'
  radius: number
}
type Shape = Square | Circle

function area(s: Shape): number {
  switch (s.kind) {
    case 'square': return s.size * s.size
    case 'circle': return Math.PI * s.radius ** 2
  }
}

6.1 函数类型 ​

tsx
// 类型表达式
let myAdd: (x: number, y: number) => number =
  function(x, y) { return x + y }

// 可选参数
function buildName(first: string, last?: string) {
  return last ? `${first} ${last}` : first
}

// 默认参数
function buildName2(first: string, last = 'Smith') { }

// 剩余参数
function sum(...nums: number[]): number {
  return nums.reduce((a, b) => a + b, 0)
}

// this 类型

  console.log(this, name)
}

6.2 函数重载 ​

tsx
function reverse(str: string): string
function reverse(arr: any[]): any[]
function reverse(strOrArr: string | any[]): string | any[] {
  if (typeof strOrArr === 'string') {
    return strOrArr.split('').reverse().join('')
  }
  return strOrArr.slice().reverse()
}

详细文档：类

7.1 基础使用 ​

tsx
class Greeter {
  greeting: string
  constructor(message: string) {
    this.greeting = message

7.2 继承与修饰符 ​

tsx
class Animal {
  name: string
  constructor(name: string) { this.name = name }
  move(distance: number = 0): void {
    console.log(`${this.name} moved ${distance}m`)
  }
}

class Dog extends Animal {
  bark(): void { console.log('Woof!') }
}

class Person {
  readonly name: string
  private age: number
  protected height: number

  constructor(name: string, age: number, height: number) {
    this.name = name
    this.age = age
    this.height = height
  }
}
修饰符含义
public公开（默认）
private私有，仅本类可访问
protected受保护，本类及子类可访问
readonly只读属性

7.3 抽象类 ​

tsx
abstract class Animal {
  abstract makeSound(): void
  move(): void { console.log('moving') }
}

class Dog extends Animal {
  makeSound(): void { console.log('bark') }
}

7.4 高级技巧 ​

九、装饰器 ​

详细文档：装饰器

9.1 类装饰器 ​

tsx
function sealed(constructor: Function) {
  Object.seal(constructor)
  Object.seal(constructor.prototype)
}

@sealed
class BugReport {
  type = 'report'
  title: string
}

9.2 方法装饰器 ​

tsx
function readonly(target: any, key: string, descriptor: PropertyDescriptor) {
  descriptor.writable = false
  return descriptor
}

class Person {
  @readonly
  name() { return 'John' }
}

9.3 属性装饰器 ​

tsx
function log(target: any, key: string) {
  let value = target[key]
  const getter = () => {
    console.log(`Getting ${key}`)
    return value
  }
  const setter = (newVal: any) => {
    console.log(`Setting ${key} to ${newVal}`)
    value = newVal
  }
  Object.defineProperty(target, key, { get: getter, set: setter })
}

  return <input value={value} onChange={handleChange} />
}

十四、常见面试题 ​

详细文档：TypeScript vs JavaScript

Q1: TypeScript 和 JavaScript 的区别？ ​

方面JavaScriptTypeScript
类型系统动态类型，弱类型静态类型，强类型
编译无需编译，解释执行编译为 JS
错误检测运行时才发现编译时发现
复杂度简单学习曲线陡峭
可维护性大型项目难维护大型项目友好

Q2: any vs unknown vs never 的区别？ ​

tsx
// any - 任意类型，绕过类型检查
let a: any = 123
a.toFixed() // ok，绕过了检查

// unknown - 未知类型，必须做类型检查
let u: unknown = 123
if (typeof u === 'number') {
  u.toFixed() // ok，安全
}

// never - 从不存在的类型
function fn(): never {
  throw new Error()
}

Q3: interface 和 type 的区别？ ​

特性interfacetype
定义对象✅✅
定义函数✅✅
继承/扩展extends&
合并声明自动合并不能合并
计算属性❌✅
条件类型❌✅
tsx
// interface 自动合并（可扩展）
interface A { a: string }
interface A { b: number } // 合并后 { a: string; b: number }

// type 不能合并
type B = { a: string } & { b: number }

Q4: 如何实现一个 Pick 操作？ ​

tsx
type MyPick<T, K extends keyof T> = {
  [P in K]: T[P]
}

interface User {
  name: string
  age: number
  email: string
}

type UserPreview = MyPick<User, 'name' | 'age'>
// { name: string; age: number }

Q5: TypeScript 如何做类型守卫？ ​

tsx
// typeof
if (typeof x === 'string') { x.toUpperCase() }

// instanceof
if (pet instanceof Dog) { pet.bark() }

// 自定义谓词
function isString(s: unknown): s is string {
  return typeof s === 'string'
}

// in
if ('a' in obj) { obj.a }

Q6: 泛型约束的作用？ ​

tsx
// 约束 T 必须有 length 属性
interface Lengthwise {

}

function loggingIdentity<T extends Lengthwise>(arg: T): T {
  console.log(arg.length) // 现在保证有 length
  return arg
}

Q7: 什么是装饰器？有什么作用？ ​

tsx
// 类装饰器 - 添加属性或修改行为
function addProperty(target: any) {
  target.prototype.newProp = 'value'
}

// 方法装饰器 - 修改方法行为
function log(target: any, key: string, descriptor: PropertyDescriptor) {
  const original = descriptor.value
  descriptor.value = function(...args: any[]) {
    console.log(`${key} called with`, args)
    return original.apply(this, args)
  }
}

// 使用
@addProperty
class MyClass {
  @log
  myMethod() { }
}
