三、文档加载 + 索引(离线) ​

python
# src/rag_service/ingest.py
from pathlib import Path
from langchain_community.document_loaders import (
    PyPDFLoader,
    DirectoryLoader,
)
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_postgres import PGVector
from .config import settings

# 1. 文档加载
def load_documents(data_dir: str = "./data/docs") -> list:
    loader = DirectoryLoader(
        data_dir,
        glob="**/*.{pdf,txt,md}",
        loader_cls=PyPDFLoader,
        recursive=True,
    )
    return loader.load()

# 2. 文档分块
def split_documents(docs: list) -> list:
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=500,
        chunk_overlap=50,
        separators=["\n\n", "\n", "。", ".", " ", ""],
    )
    return splitter.split_documents(docs)

# 3. Embedding + 存入 pgvector
def build_index():
    print("加载文档...")
    docs = load_documents()
    print(f"  文档数: {len(docs)}")

    print("分块...")
    chunks = split_documents(docs)
    print(f"  分块数: {len(chunks)}")

    print("Embedding + 存储...")
    embeddings = OpenAIEmbeddings(model=settings.embedding_model)
    vectorstore = PGVector.from_documents(
        chunks,
        embeddings,
        connection_string=settings.database_url,
        collection_name="docs",
        pre_delete_collection=True,  # 重建(开发用)
    )
    print(f"  ✅ 索引完成")

if __name__ == "__main__":
    build_index()

四、检索(高级策略) ​

python
# src/rag_service/retrieve.py
from langchain_postgres import PGVector
from langchain.retrievers import (
    EnsembleRetriever,
    ContextualCompressionRetriever,
)
from langchain.retrievers.document_compressors import CrossEncoderReranker
from langchain_community.cross_encoders import HuggingFaceCrossEncoder
from langchain_openai import OpenAIEmbeddings
from .config import settings

def get_retriever():
    """获取高级检索器(混合 + 重排序 + 压缩)"""
    embeddings = OpenAIEmbeddings(model=settings.embedding_model)
    vectorstore = PGVector.from_existing_index(
        embeddings,
        connection_string=settings.database_url,
        collection_name="docs",
    )

    # 1. 基础检索:Top-K
    base_retriever = vectorstore.as_retriever(
        search_type="mmr",  # 多样性
        search_kwargs={
            "k": settings.top_k * 2,  # 取 2 倍候选,再重排序
            "fetch_k": 20,
        },
    )

    # 2. 重排序(精排到 Top-K)
    reranker = HuggingFaceCrossEncoder(
        model_name="BAAI/bge-reranker-large",
        model_kwargs={"device": "cuda"},
    )
    compression_retriever = ContextualCompressionRetriever(
        base_compressor=CrossEncoderReranker(model=reranker, top_n=settings.top_k),
        base_retriever=base_retriever,
    )

    return compression_retriever

五、RAG Chain(LCEL) ​

python
# src/rag_service/chain.py
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough
from .config import settings
from .retrieve import get_retriever

# Prompt
PROMPT_TEMPLATE = """你是企业知识助手。基于以下参考资料回答问题。

要求:
1. 只基于参考资料回答
2. 如果参考资料不足,说"抱歉,我没有找到相关信息"
3. 引用时标注来源(文档名 + 段落)
4. 回答用中文,简洁清晰

参考资料:
{context}

问题:{question}

回答:"""

prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
llm = ChatOpenAI(model=settings.llm_model, temperature=0)

def format_docs(docs) -> str:
    """把检索到的文档格式化"""
    formatted = []
    for i, doc in enumerate(docs, 1):
        source = doc.metadata.get("source", "未知")
        formatted.append(f"[{i}] (来源:{source})\n{doc.page_content}")
    return "\n\n".join(formatted)

# 完整 Chain
retriever = get_retriever()
rag_chain = (
    {
        "context": retriever | format_docs,
        "question": RunnablePassthrough(),
    }
    | prompt
    | llm
    | StrOutputParser()
)

# 同步调用
def ask(question: str) -> str:
    return rag_chain.invoke(question)

# 流式调用
def ask_stream(question: str):
    for chunk in rag_chain.stream(question):
        yield chunk

# 带源文档(用于显示来源)
def ask_with_sources(question: str) -> dict:
    """返回答案 + 来源文档"""
    docs = retriever.invoke(question)
    context = format_docs(docs)
    answer = (prompt | llm | StrOutputParser()).invoke({
        "context": context,
        "question": question,
    })
    return {
        "answer": answer,
        "sources": [
            {"content": d.page_content, "metadata": d.metadata}
            for d in docs
        ],
    }

六、FastAPI 部署 ​

# src/rag_service/api.py
from fastapi import FastAPI, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from .chain import ask, ask_stream, ask_with_sources

app = FastAPI(title="RAG Service")

class Question(BaseModel):
    question: str
    user_id: str | None = None

class Answer(BaseModel):
    answer: str
    sources: list[dict] | None = None

@app.post("/ask", response_model=Answer)
async def ask_question(req: Question):
    try:
        if req.user_id:
            result = ask_with_sources(req.question)
            return Answer(**result)
        else:
            answer = ask(req.question)
            return Answer(answer=answer)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/ask/stream")
async def ask_stream_question(req: Question):
    """流式响应"""
    def generate():
        for chunk in ask_stream(req.question):
            yield chunk

    return StreamingResponse(generate(), media_type="text/plain")

python
# src/rag_service/evaluate.py
from datasets import Dataset
from ragas import evaluate
from ragas.metrics import (
    faithfulness,
    answer_relevancy,
    context_precision,
    context_recall,
)
from .chain import ask

# 1. 准备测试集
test_data = {
    "question": [
        "年假政策是什么?",
        "病假可以请几天?",
        "加班工资怎么算?",
    ],
    "ground_truth": [
        "员工每年有 10 天年假",
        "病假最多 5 天",
        "加班 1.5 倍工资",
    ],
}

# 2. 跑 RAG,收集 answer + contexts
results = {"question": [], "answer": [], "contexts": [], "ground_truth": []}
for q, gt in zip(test_data["question"], test_data["ground_truth"]):
    response = ask_with_sources(q)
    results["question"].append(q)
    results["answer"].append(response["answer"])
    results["contexts"].append([s["content"] for s in response["sources"]])
    results["ground_truth"].append(gt)

# 3. 评估
dataset = Dataset.from_dict(results)
scores = evaluate(
    dataset,
    metrics=[faithfulness, answer_relevancy, context_precision, context_recall],
)

print(scores)
# {'faithfulness': 0.95, 'answer_relevancy': 0.88, ...}
