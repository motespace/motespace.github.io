LangChain Prompt 模板与输出解析 ​

LangChain 提供结构化的 Prompt 管理 + 输出解析。本章系统讲解两个核心能力。

一、PromptTemplate 详解 ​

1.1 基础模板 ​

python
from langchain_core.prompts import PromptTemplate

# 方式 1:from_template
template = PromptTemplate.from_template("""
你是 {role}。

任务:{task}

输入:{input}
输出:
""")

# 渲染
prompt_text = template.format(
    role="Python 教师",
    task="解释概念",
    input="什么是装饰器?",
)
print(prompt_text)

# invoke 方式(返回 StringPromptValue 对象)
prompt_value = template.invoke({
    "role": "Python 教师",
    "task": "解释概念",
    "input": "什么是装饰器?",
})

1.2 ChatPromptTemplate(多角色) ​

python
from langchain_core.prompts import ChatPromptTemplate

# 1. 多消息模板
template = ChatPromptTemplate.from_messages([
    ("system", "你是{role},回答用中文,长度不超过{length}字。"),
    ("human", "{input}"),
])

messages = template.format_messages(
    role="Python 教师",
    length=200,
    input="什么是装饰器?",
)

# 2. 占位符(动态消息列表)
template_with_history = ChatPromptTemplate.from_messages([
    ("system", "你是友好的助手。"),
    ("placeholder", "{chat_history}"),  # 历史对话占位符
    ("human", "{input}"),
])

messages = template_with_history.format_messages(
    chat_history=[
        ("human", "你好"),
        ("ai", "你好!有什么可以帮你?"),
    ],
    input="Python 是什么?",
)

1.3 FewShotChatMessagePromptTemplate ​

python
from langchain_core.prompts import FewShotChatMessagePromptTemplate, ChatPromptTemplate

examples = [
    {"input": "什么是 Python?", "output": "编程语言"},
    {"input": "什么是 JavaScript?", "output": "Web 脚本语言"},
    {"input": "什么是 Rust?", "output": "系统编程语言"},
]

example_prompt = ChatPromptTemplate.from_messages([
    ("human", "{input}"),
    ("ai", "{output}"),
])

few_shot_prompt = FewShotChatMessagePromptTemplate(
    example_prompt=example_prompt,
    examples=examples,
)

# 组合
final_prompt = ChatPromptTemplate.from_messages([
    ("system", "你是技术词典。用一句话回答。"),
    few_shot_prompt,
    ("human", "{input}"),
])

messages = final_prompt.format_messages(input="什么是 Go?")

二、Message 类型 ​

python
from langchain_core.messages import (
    HumanMessage,
    AIMessage,
    SystemMessage,
    FunctionMessage,
    ToolMessage,
)

# 1. HumanMessage(用户输入)
msg = HumanMessage(content="你好")

# 2. AIMessage(AI 回复)
msg = AIMessage(content="你好!有什么可以帮你?")

# 3. SystemMessage(系统设定)
msg = SystemMessage(content="你是友好的助手")

# 4. 带附加信息
msg = HumanMessage(
    content="你好",
    name="user_123",  # 用户标识(多用户场景)
)

# 5. FunctionMessage(旧版,Function Calling 结果)
msg = FunctionMessage(
    name="get_weather",
    content='{"temp": 25}',
)

# 6. ToolMessage(新版,Tool Use 结果)
msg = ToolMessage(
    content='{"temp": 25}',
    tool_call_id="call_xxx",
)

四、OutputParser 详解 ​

4.1 字符串解析(默认) ​

python
from langchain_core.output_parsers import StrOutputParser

parser = StrOutputParser()
chain = llm | parser
result = chain.invoke([HumanMessage(content="讲个笑话")])
print(result)  # 字符串

4.2 JSON 解析 ​

python
from langchain_core.output_parsers import JsonOutputParser
from pydantic import BaseModel, Field

class PersonInfo(BaseModel):
    name: str = Field(..., description="姓名")
    age: int = Field(..., description="年龄")
    occupation: str = Field(..., description="职业")

parser = JsonOutputParser(pydantic_object=PersonInfo)

prompt = ChatPromptTemplate.from_messages([
    ("system", "提取人物信息。\n{format_instructions}"),
    ("human", "{input}"),
])

chain = prompt.partial(
    format_instructions=parser.get_format_instructions()
) | llm | parser

result = chain.invoke({"input": "张三,28 岁,Python 工程师"})
print(result)  # PersonInfo(name='张三', age=28, occupation='Python 工程师')

4.3 自定义解析器 ​

python
from langchain_core.output_parsers import BaseOutputParser
from langchain_core.messages import BaseMessage

class CommaSeparatedListParser(BaseOutputParser[list[str]]):
    """解析逗号分隔的列表"""

    def parse_result(self, result, *, partial: bool = False) -> list[str]:
        text = result.content if hasattr(result, "content") else str(result)
        return [item.strip() for item in text.split(",") if item.strip()]

    @property
    def _type(self) -> str:
        return "comma_list"

# 用
parser = CommaSeparatedListParser()
chain = prompt | llm | parser

result = chain.invoke({"input": "水果"})
# ["苹果", "梨", "香蕉", "葡萄"]

4.4 重试 + 容错解析 ​

五、Structured Output(OpenAI 官方) ​

python
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

class MovieReview(BaseModel):
    title: str = Field(..., description="电影名")
    rating: int = Field(..., ge=1, le=10, description="评分 1-10")
    genres: list[str] = Field(..., description="类型列表")

llm = ChatOpenAI(model="gpt-4o-2024-08-06")

# 强制 LLM 输出 MovieReview 结构
structured_llm = llm.with_structured_output(MovieReview)

result = structured_llm.invoke("""
《盗梦空间》是 2010 年诺兰导演的科幻电影,IMDb 评分 8.8,
类型包括科幻、动作、悬疑。
""")

print(result.title, result.rating, result.genres)
# 盗梦空间 9 ['科幻', '动作', '悬疑']

六、实战:完整的 Prompt + Parser 链 ​

python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import JsonOutputParser
from pydantic import BaseModel, Field

class CodeReview(BaseModel):
    score: int = Field(..., ge=0, le=100, description="代码质量评分")
    issues: list[str] = Field(..., description="问题列表")
    suggestions: list[str] = Field(..., description="建议列表")

llm = ChatOpenAI(model="gpt-4o")

prompt = ChatPromptTemplate.from_messages([
    ("system", """
你是资深 Python 代码审查专家。

审查代码,输出 JSON:
{format_instructions}

要求:
- score: 0-100
- issues: 每条问题一句话描述
- suggestions: 具体修复建议
"""),
    ("human", "代码:\n```python\n{code}\n```"),
])

parser = JsonOutputParser(pydantic_object=CodeReview)

chain = prompt.partial(
    format_instructions=parser.get_format_instructions()
) | llm | parser

result = chain.invoke({
    "code": """
def get_users():
    return db.query("SELECT * FROM users")
"""
})

print(f"评分: {result['score']}")
for issue in result['issues']:
    print(f"问题: {issue}")
for suggestion in result['suggestions']:
    print(f"建议: {suggestion}")

七、动态 Prompt(根据上下文调整) ​

python
from datetime import datetime

def get_dynamic_system_prompt(user_context: dict) -> str:
    """根据用户上下文动态生成"""
    return f"""你是 AI 助手。

当前用户:
- 角色:{user_context.get('role', '匿名')}
- 偏好:{user_context.get('preferences', '专业')}
- 时间:{datetime.now().strftime('%Y-%m-%d %H:%M')}

根据用户角色调整回答:
- 开发者:给代码示例
- 产品:关注业务价值
- 新手:用通俗语言"""

# 动态绑定
def chat(user_context: dict, question: str):
    prompt = ChatPromptTemplate.from_messages([
        ("system", get_dynamic_system_prompt(user_context)),
        ("human", "{question}"),
    ])
    chain = prompt | llm | StrOutputParser()
    return chain.invoke({"question": question})

🎯 面试常考 ​

Q1：PromptTemplate 和 ChatPromptTemplate 的区别?

前者渲染成字符串,后者渲染成消息数组(能区分 system/user/assistant),Chat 模型必须用后者。

Q2：Few-shot 示例放在哪里?

放在 system 或 user 消息的前面,作为「示例」段落。示例要覆盖边界情况,数量 2~5 个通常就够,太多会挤占 token 且收益递减。

Q3：怎么复用 Prompt 的一部分?

用 partial:prompt.partial(language='中文') 得到一个已固定部分变量的模板,常用于多语言/多角色场景。
