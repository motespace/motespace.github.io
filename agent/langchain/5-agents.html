LangChain Agent + Tool ​

Agent = 让 LLM 自主决定调用哪些工具,完成复杂任务。 本章系统讲解 LangChain Agent 实战。

一、Agent 是什么? ​

传统 Chain:固定流程(步骤 1 → 2 → 3)
Agent:动态决策(LLM 自己决定下一步)

例:
用户:"北京今天天气怎么样,适合穿什么?"

❌ Chain(固定):必须按"查天气 → 查穿衣建议"执行
✅ Agent:LLM 可能决定:
   1. 先调用 get_weather("北京")
   2. 基于结果,决定要不要查"穿衣建议"
   3. 也可能再查"今天空气质量"
   4. 综合所有结果回复用户

二、Tool 定义 ​

python
from langchain_core.tools import tool

# 方式 1:用 @tool 装饰器(自动从函数签名生成 schema)
@tool
def get_weather(location: str, unit: str = "celsius") -> str:
    """查询指定城市的天气信息,包括温度、湿度、天气状况。

    当用户问"XX 城市天气怎么样"时调用此工具。
    """
    # 实际调用天气 API
    return f"{location}:晴,25°C,湿度 60%"

@tool
def search_docs(query: str, max_results: int = 3) -> str:
    """在技术文档库中搜索相关文档。
    
    Args:
        query: 搜索关键词
        max_results: 最多返回几条结果
    """
    # 实际搜索逻辑
    return f"找到 {max_results} 条关于 '{query}' 的文档"

@tool
def calculate(expression: str) -> str:
    """计算数学表达式,例如 '2 + 3 * 4'"""
    try:
        return str(eval(expression))
    except Exception as e:
        return f"Error: {e}"

# 工具列表
tools = [get_weather, search_docs, calculate]
python
# 方式 2:StructuredTool(更精细控制)
from langchain_core.tools import StructuredTool
from pydantic import BaseModel, Field

class WeatherInput(BaseModel):
    location: str = Field(..., description="城市名")
    unit: str = Field("celsius", description="温度单位")

def get_weather(location: str, unit: str = "celsius") -> str:
    return f"{location}:晴,25°C"

weather_tool = StructuredTool.from_function(
    func=get_weather,
    name="get_weather",
    description="查询天气",
    args_schema=WeatherInput,
)
python
# 方式 3:从 OpenAI Function schema 创建
from langchain_core.tools import Tool

weather_tool = Tool(
    name="get_weather",
    description="查询天气信息",
    func=lambda location: f"{location}:晴,25°C",
)

三、Agent 创建(三种方式) ​

3.1 OpenAI Functions Agent(推荐) ​

python
from langchain_openai import ChatOpenAI
from langchain.agents import create_tool_calling_agent, AgentExecutor
from langchain_core.prompts import ChatPromptTemplate

llm = ChatOpenAI(model="gpt-4o")

# 1. Prompt
prompt = ChatPromptTemplate.from_messages([
    ("system", "你是助手,可以使用工具回答用户问题。"),
    ("placeholder", "{chat_history}"),
    ("human", "{input}"),
    ("placeholder", "{agent_scratchpad}"),  # Agent 中间步骤
])

# 2. 创建 Agent
agent = create_tool_calling_agent(llm, tools, prompt)

# 3. 创建 Executor
agent_executor = AgentExecutor(
    agent=agent,
    tools=tools,
    verbose=True,  # 打印推理过程
    max_iterations=5,  # 最多 5 轮
    handle_parsing_errors=True,  # 解析错误时重试
)

# 4. 调用
result = agent_executor.invoke({"input": "北京今天天气怎么样?"})
print(result["output"])

四、完整实战 ​

python
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.tools import tool, create_retriever_tool
from langchain.agents import create_tool_calling_agent, AgentExecutor
from langchain_core.prompts import ChatPromptTemplate

# 1. 准备工具
@tool
def get_current_time() -> str:
    """获取当前时间"""
    from datetime import datetime
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

@tool
def calculate(expression: str) -> str:
    """计算数学表达式"""
    return str(eval(expression))

# 2. RAG 工具
docs = ["Python 是编程语言", "LangChain 是 LLM 框架", "RAG 是检索增强生成"]
vectorstore = FAISS.from_texts(docs, OpenAIEmbeddings())
retriever = vectorstore.as_retriever(search_kwargs={"k": 1})

retriever_tool = create_retriever_tool(
    retriever,
    name="search_knowledge_base",
    description="搜索技术知识库,当用户问技术问题时调用",
)

tools = [get_current_time, calculate, retriever_tool]

# 3. 创建 Agent
llm = ChatOpenAI(model="gpt-4o")

prompt = ChatPromptTemplate.from_messages([
    ("system", """你是技术助手,可以:
- 用 search_knowledge_base 查技术资料
- 用 calculate 做计算
- 用 get_current_time 查时间

回答问题前,先看看是否需要工具。"""),
    ("placeholder", "{chat_history}"),
    ("human", "{input}"),
    ("placeholder", "{agent_scratchpad}"),
])

agent = create_tool_calling_agent(llm, tools, prompt)
agent_executor = AgentExecutor(
    agent=agent,
    tools=tools,
    verbose=True,
    max_iterations=5,
)

# 4. 使用
questions = [
    "Python 是什么?",
    "1+1 等于几?",
    "现在几点?",
    "LangChain 是做什么的?",
]

for q in questions:
    print(f"\n{'=' * 50}\nQ: {q}\n{'=' * 50}")
    result = agent_executor.invoke({"input": q})
    print(f"A: {result['output']}")

五、流式输出 ​

python
# Agent 也支持流式
async def stream_agent():
    async for event in agent_executor.astream_events({"input": "你好"}, version="v2"):
        kind = event["event"]

        if kind == "on_tool_start":
            print(f"\n🔧 调用工具: {event['name']}")
        elif kind == "on_tool_end":
            print(f"✅ 工具返回: {event['data'].get('output', '')[:100]}")
        elif kind == "on_chat_model_stream":
            content = event["data"]["chunk"].content
            if content:
                print(content, end="", flush=True)
        elif kind == "on_chain_end":
            print(f"\n✅ 最终输出: {event['data']['output']}")

asyncio.run(stream_agent())

八、错误处理 ​

python
from langchain.agents import AgentExecutor

agent_executor = AgentExecutor(
    agent=agent,
    tools=tools,
    verbose=True,
    max_iterations=5,
    max_execution_time=30,  # 30 秒超时
    early_stopping_method="generate",  # 强制结束策略
    handle_parsing_errors=True,  # 解析失败重试
    return_intermediate_steps=True,  # 返回中间步骤
)

# 捕获异常
try:
    result = agent_executor.invoke({"input": "复杂问题"})
except Exception as e:
    print(f"Agent 失败: {e}")

# 中间步骤
result = agent_executor.invoke({"input": "..."})
for step in result["intermediate_steps"]:
    action, observation = step
    print(f"调用: {action.tool}({action.tool_input})")
    print(f"结果: {observation[:100]}\n")

九、Agent vs Chain 选型 ​

用 Chain(固定流程):
✅ 流程明确,每步都该做
✅ 性能要求高(避免 LLM 决策 overhead)
✅ 可预测

用 Agent(动态决策):
✅ 任务多变,需要灵活
✅ 需要 LLM 自主决策
✅ 工具数量多(>3)

混合:
✅ 大部分步骤固定 + 部分步骤决策
✅ Plan-and-Execute(先用 LLM 规划,再固定执行)

class AgentDebugger(BaseCallbackHandler):
    def on_agent_action(self, action, **kwargs):
