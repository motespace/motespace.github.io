# 实现一个带缓存的装饰器
def memoize(func):
    cache = {}
    @functools.wraps(func)
    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]
    return wrapper

import aiohttp

async def fetch_url(session, url):
    async with session.get(url) as response:
        return await response.text()

async def fetch_all(urls):
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_url(session, url) for url in urls]
        return await asyncio.gather(*tasks)

Day 14｜RAG 完整架构 ​

目标：搭建完整的 RAG 系统

知识点清单 ​

  • RAG 三阶段

    离线索引：
文档 → 分块 → Embedding → 向量数据库

在线检索：
用户问题 → Embedding → 相似度检索 → Top-K 文档 → Prompt 组装 → LLM → 答案

  • 检索优化 ⭐

    • Reranking：初检 → 重排序（用更强的模型）
    • HyDE：假设文档嵌入
    • 多路召回：向量检索 + 关键词检索（BM25）
    • Query 改写：让问题更适合检索

  • RAG 评估

    • 答案相关性（Answer Relevancy）
    • 上下文相关性（Context Relevancy）
    • 忠实度（Faithfulness）
    • 工具：ragas 框架

  • 高级模式

    • GraphRAG（微软）：基于知识图谱
    • Self-RAG：模型自己决定是否检索
    • Agentic RAG：Agent 自主选择检索策略

实践任务 ​

python
# 最小 RAG 实现
from openai import OpenAI
import chromadb

def simple_rag(question: str, collection):
    # 1. 检索
    results = collection.query(query_texts=[question], n_results=3)
    context = "\n".join(results["documents"][0])

    # 2. Prompt 组装
    prompt = f"""基于以下资料回答问题。如果资料中没有，请说"我不知道"。

资料：
{context}

问题：{question}
答案："""

    # 3. LLM 生成
    response = client.chat.completions.create(
        model="deepseek-chat",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

Day 17-18｜LangChain RAG 实战 ​

目标：用 LangChain 重构项目 A

知识点清单 ​

  • RAG 核心类

    • DocumentLoader：文档加载
    • TextSplitter：分块
    • Embeddings：向量化
    • VectorStore：向量存储
    • Retriever：检索器接口

  • 完整 RAG Chain

    python
    from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_community.vectorstores import Chroma
from langchain_core.runnables import RunnablePassthrough
from langchain_core.prompts import ChatPromptTemplate

# 1. 加载文档
loader = PyPDFLoader("doc.pdf")
docs = loader.load()

# 2. 分块
splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
splits = splitter.split_documents(docs)

# 3. 向量化 + 存储
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
vectorstore = Chroma.from_documents(splits, embeddings)

# 4. 检索器
retriever = vectorstore.as_retriever(search_kwargs={"k": 3})

# 5. Prompt
prompt = ChatPromptTemplate.from_template("""
基于上下文回答问题：
{context}

问题：{question}
""")

# 6. 完整 Chain
llm = ChatOpenAI(model="deepseek-chat")
rag_chain = (
    {"context": retriever, "question": RunnablePassthrough()}
    | prompt
    | llm
    | StrOutputParser()
)

result = rag_chain.invoke("文档讲了什么？")

  • 进阶检索器

    • MultiQueryRetriever：多角度检索
    • ContextualCompressionRetriever：压缩无关内容
    • EnsembleRetriever：混合检索（向量 + BM25）

    from langchain.tools import tool

@tool
def search_web(query: str) -> str:

def calculate(expression: str) -> str:

  • 使用 Agent

Day 21-22｜LangGraph 入门到进阶 ​

目标：构建复杂的 Agent 工作流

知识点清单 ​

  • 为什么需要 LangGraph

    • LangChain Agent：线性循环，难处理复杂流程
    • LangGraph：图结构，可以条件分支、循环、子图
    • 适合：多步骤、有状态、需要人工介入的流程

  • 核心概念

    python
    from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated
from langgraph.graph.message import add_messages

# 1. 定义状态
class State(TypedDict):
    messages: Annotated[list, add_messages]

# 2. 定义节点
def chatbot(state: State):
    response = llm.invoke(state["messages"])
    return {"messages": [response]}

# 3. 构建图
graph = StateGraph(State)
graph.add_node("chatbot", chatbot)
graph.add_edge("chatbot", END)
graph.set_entry_point("chatbot")

# 4. 编译运行
app = graph.compile()
result = app.invoke({"messages": [("user", "你好")]})

  • 核心特性 ⭐

    • State：图的状态（核心概念）
    • Nodes：执行单元
    • Edges：流转关系
    • Conditional Edges：条件分支
    • Cycles：循环（Agent 的关键）
    • Persistence：状态持久化（Checkpoint）
    • Human-in-the-Loop：人工介入

  • 条件边示例

    python
    def should_continue(state: State) -> str:
    last_message = state["messages"][-1]
    if last_message.tool_calls:
        return "tools"
    return END

graph.add_conditional_edges(
    "agent",
    should_continue,
    {"tools": "tool_node", END: END}
)

  • 持久化与记忆

    python
    from langgraph.checkpoint.memory import MemorySaver

memory = MemorySaver()
app = graph.compile(checkpointer=memory)

# 用 thread_id 区分会话
config = {"configurable": {"thread_id": "user-123"}}

    • 适合：代码生成、写作

    • Supervisor 模式：一个主管协调多个专家
    • 对等模式：Agent 之间直接对话
    • 框架：LangGraph / CrewAI / AutoGen

    • Anthropic 提出的协议

    • 快速接入第三方服务

    from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

app = FastAPI()

class ChatRequest(BaseModel):
    message: str
    conversation_id: str | None = None

@app.post("/chat/stream")
async def chat_stream(request: ChatRequest):
    async def generate():
        stream = client.chat.completions.create(
            model="deepseek-chat",
            messages=[{"role": "user", "content": request.message}],
            stream=True
        )
        for chunk in stream:
            if chunk.choices[0].delta.content:
                yield f"data: {chunk.choices[0].delta.content}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(generate(), media_type="text/event-stream")
