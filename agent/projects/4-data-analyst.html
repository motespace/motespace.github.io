项目 4:数据分析 Agent(Text-to-SQL + 图表) ​

自然语言提问 → 自动生成 SQL → 查询数据库 → 生成可视化图表 + 洞察报告。 Text-to-SQL + 数据可视化,业务团队的"AI 数据分析师"。

一、需求 ​

场景:业务人员想查数据,但不会 SQL
例:
  "上个月北京地区新注册用户数?"
  "对比 Q1 和 Q2 各产品线销售?"
  "哪个渠道获客成本最低?"

传统:
  - 找数据团队(慢)
  - 学 SQL(门槛高)

解决方案:数据分析 Agent
  - 自然语言 → SQL(LLM 自动生成)
  - 自动执行 + 返回结果
  - 自动生成图表 + 洞察

二、架构 ​

用户问题
   ↓
SQL Generator → 生成 SQL(参考表结构)
   ↓
SQL Validator → 检查 SQL 安全性
   ↓
SQL Executor → 执行查询
   ↓
Data Analyzer → 分析结果,生成洞察
   ↓
Chart Recommender → 推荐图表类型
   ↓
Final Output(SQL + 结果 + 图表 + 洞察)

四、核心代码 ​

python
# src/agent.py
from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver
from langgraph.graph.message import add_messages
from langchain_openai import ChatOpenAI
from langchain_core.messages import SystemMessage, HumanMessage
from langchain_community.utilities import SQLDatabase
from langchain_community.tools.sql_database.tool import QuerySQLDataBaseTool
import json

llm = ChatOpenAI(model="gpt-4o")

# ===== 数据库连接 =====
db = SQLDatabase.from_uri("postgresql://user:pass@localhost:5432/ecommerce")

# 让 LLM 知道表结构(schema)
table_info = db.get_table_info()
# "CREATE TABLE users (...) \n CREATE TABLE orders (...)..."

# ===== 工具 =====
sql_tool = QuerySQLDataBaseTool(db=db)

# ===== Agent =====
class AnalysisState(TypedDict):
    question: str
    sql: str
    result: str
    chart_type: str
    insights: str
    messages: Annotated[list, add_messages]

def generate_sql(state: AnalysisState):
    """生成 SQL"""
    response = llm.invoke([
        SystemMessage(content=f"""你是 SQL 专家。
基于以下 schema 生成 SQL:

{table_info}

要求:
- 只用 SELECT,不用 INSERT/UPDATE/DELETE
- 加 LIMIT 100(防查询过大)
- 用 ORDER BY 排序,如果有"top"则降序

只输出 SQL。"""),
        HumanMessage(content=state["question"]),
    ])

    sql = response.content.strip()
    # 清理 markdown 标记
    sql = sql.replace("```sql", "").replace("```", "").strip()

    return {"sql": sql, "messages": [response]}

def execute_sql(state: AnalysisState):
    """执行 SQL"""
    try:
        result = sql_tool.invoke(state["sql"])
        return {"result": str(result)}
    except Exception as e:
        return {"result": f"SQL 错误: {e}"}

def recommend_chart(state: AnalysisState):
    """推荐图表类型"""
    response = llm.invoke([
        SystemMessage(content="""根据查询结果,推荐图表类型:
- 单一数值 → 'number'
- 时间序列 → 'line'
- 类别对比 → 'bar'
- 占比 → 'pie'
- 关系 → 'scatter'

只输出图表类型名。"""),
        HumanMessage(content=f"SQL:{state['sql']}\n结果:{state['result']}"),
    ])
    return {"chart_type": response.content.strip().lower()}

def generate_insights(state: AnalysisState):
    """生成洞察"""
    response = llm.invoke([
        SystemMessage(content="""你是数据分析师。基于 SQL 结果给出 3-5 条关键洞察。

要求:
- 具体数字(不要泛泛而谈)
- 商业价值(对决策的意义)
- 风险 / 机会
- 中文"""),
        HumanMessage(content=f"""
问题:{state['question']}
SQL:{state['sql']}
结果:{state['result']}
图表:{state['chart_type']}

生成洞察报告。"""),
    ])
    return {"insights": response.content}

# ===== 构建图 =====
builder = StateGraph(AnalysisState)
builder.add_node("generate_sql", generate_sql)
builder.add_node("execute_sql", execute_sql)
builder.add_node("recommend_chart", recommend_chart)
builder.add_node("generate_insights", generate_insights)

builder.add_edge("generate_sql", "execute_sql")
builder.add_edge("execute_sql", "recommend_chart")
builder.add_edge("recommend_chart", "generate_insights")
builder.add_edge("generate_insights", END)

graph = builder.compile(checkpointer=MemorySaver())

# ===== 使用 =====
config = {"configurable": {"thread_id": "user_123"}}

result = graph.invoke(
    {"question": "上个月北京地区新注册用户数?"},
    config=config,
)

print(f"SQL:{result['sql']}")
print(f"结果:{result['result']}")
print(f"图表:{result['chart_type']}")
print(f"洞察:\n{result['insights']}")

五、安全:只允许 SELECT ​

python
def execute_sql_safe(state: AnalysisState):
    """执行 SQL(严格白名单)"""
    sql = state["sql"].strip().upper()

    # 只允许 SELECT / WITH(CTE)
    if not (sql.startswith("SELECT") or sql.startswith("WITH")):
        return {"result": f"❌ 不允许的 SQL 类型:{sql[:20]}"}

    # 禁止危险关键字
    forbidden = ["DROP", "DELETE", "UPDATE", "INSERT", "TRUNCATE", "ALTER", "GRANT", "REVOKE"]
    for keyword in forbidden:
        if keyword in sql:
            return {"result": f"❌ 包含禁用关键字:{keyword}"}

    # 加 LIMIT 防护
    if "LIMIT" not in sql:
        sql += " LIMIT 100"

    try:
        result = sql_tool.invoke(sql)
        return {"result": str(result)}
    except Exception as e:
        return {"result": f"SQL 错误: {e}"}

# 1. Schema 缓存(避免每次重新读取)
from functools import lru_cache

@lru_cache(maxsize=1)
def get_table_info():
    return db.get_table_info()

# 2. SQL 模板(加速常见查询)
COMMON_TEMPLATES = {
    "用户数": "SELECT COUNT(*) FROM users WHERE ...",
    "销售额": "SELECT SUM(amount) FROM orders WHERE ...",
}

def try_template(question: str) -> str | None:
    for key, sql in COMMON_TEMPLATES.items():
        if key in question:
            return sql
    return None

# 3. 慢查询监控 + 缓存
import time

def execute_sql_monitored(state):
    start = time.time()
    result = execute_sql_safe(state)
    elapsed = time.time() - start

    if elapsed > 5:
        logging.warning(f"慢查询({elapsed:.1f}s):{state['sql']}")

    return result
