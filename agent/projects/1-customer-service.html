项目 1:AI 客服系统(RAG + 工具调用) ​

第一个完整 AI 项目。从需求到部署,做一个真实可用的客服系统。 知识库问答 + 订单查询 + 工单创建 + 流式响应。

一、需求分析 ​

场景:电商公司客服
用户问题类型:
  1. 知识类(退货政策 / 物流时效 / 优惠券规则)→ RAG 查询
  2. 操作类(查我的订单 / 申请退款)→ 工具调用
  3. 闲聊类(你好 / 你是谁)→ 直接回答

技术需求:
  ✅ 多轮对话(记住上文)
  ✅ 流式响应(token 级)
  ✅ 来源追溯(显示参考文档)
  ✅ 工具调用(查订单 / 创建工单)
  ✅ 简单前端(聊天界面)

五、RAG 模块(知识库检索) ​

python
# src/rag_service.py
from pathlib import Path
from langchain_community.document_loaders import DirectoryLoader, PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_postgres import PGVector

class KnowledgeBase:
    """知识库管理"""

    def __init__(self, db_url: str):
        self.embeddings = OpenAIEmbeddings()
        self.vectorstore = PGVector.from_existing_index(
            self.embeddings,
            connection_string=db_url,
            collection_name="knowledge",
        )

    def build_index(self, docs_dir: str = "./data/docs"):
        """构建知识库索引(运行一次)"""
        loader = DirectoryLoader(
            docs_dir,
            glob="**/*.md",  # 也支持 *.pdf
            recursive=True,
        )
        docs = loader.load()

        # 分块
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=500,
            chunk_overlap=50,
        )
        chunks = splitter.split_documents(docs)

        # Embedding + 存储
        PGVector.from_documents(
            chunks,
            self.embeddings,
            connection_string=self.db_url,
            collection_name="knowledge",
            pre_delete_collection=True,
        )

    def search(self, query: str, k: int = 3) -> list[dict]:
        """检索相关文档"""
        docs = self.vectorstore.similarity_search_with_score(query, k=k)
        return [
            {
                "content": doc.page_content,
                "source": doc.metadata.get("source", "未知"),
                "score": float(score),
            }
            for doc, score in docs
        ]

# src/tools.py
from langchain_core.tools import tool
from sqlalchemy import create_engine
import json

engine = create_engine(settings.database_url)

@tool
def get_user_orders(user_email: str) -> str:
    """查询用户的所有订单。"""
    with engine.connect() as conn:
        result = conn.execute(
            "SELECT order_no, status, total FROM orders "
            "WHERE user_id = (SELECT id FROM users WHERE email = %s)",
            (user_email,),
        )
        orders = [
            {"order_no": row[0], "status": row[1], "total": float(row[2])}
            for row in result
        ]
    return json.dumps(orders, ensure_ascii=False)

@tool
def get_order_detail(order_no: str) -> str:
    """查询订单详情,包括物流信息。"""
    with engine.connect() as conn:
        result = conn.execute(
            "SELECT * FROM orders WHERE order_no = %s",
            (order_no,),
        ).fetchone()
    if not result:
        return "订单不存在"

    return json.dumps({
        "order_no": result[1],
        "status": result[2],
        "total": float(result[3]),
        "created_at": str(result[4]),
    }, ensure_ascii=False)

@tool
def create_ticket(user_email: str, title: str, content: str) -> str:
    """创建客服工单。"""
    with engine.connect() as conn:
        result = conn.execute(
            "INSERT INTO tickets (user_id, title, content) "
            "VALUES ((SELECT id FROM users WHERE email = %s), %s, %s) "
            "RETURNING id",
            (user_email, title, content),
        ).fetchone()
    return f"工单 #{result[0]} 已创建,我们会尽快处理"

七、Agent 模块(LangGraph) ​

python
# src/agent.py
from typing import TypedDict, Annotated, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import ToolNode
from langgraph.checkpoint.postgres import PostgresSaver
from langgraph.graph.message import add_messages
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage, AIMessage
from langchain_core.prompts import ChatPromptTemplate

from .rag_service import KnowledgeBase
from .tools import get_user_orders, get_order_detail, create_ticket

# ===== State =====
class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    user_email: str
    used_rag: bool
    used_tools: bool

# ===== 系统 Prompt =====
SYSTEM_PROMPT = """你是{name}的 AI 客服助手。

你的职责:
1. 回答用户关于产品、政策的问题(用知识库)
2. 帮用户查询订单、申请退款(用工具)
3. 友好、专业、简洁

回复要求:
- 中文,简洁
- 如用知识库,标注来源
- 如用工具,清晰展示结果
"""

# ===== Nodes =====
def classify_intent(state: AgentState) -> dict:
    """分类用户意图"""
    last_msg = state["messages"][-1].content
    response = ChatOpenAI(model="gpt-4o-mini").invoke([
        SystemMessage(content="""判断用户意图:
- 'rag': 知识类问题(政策 / 物流 / 优惠)
- 'tool': 操作类(查订单 / 申请退款)
- 'chat': 闲聊

只输出一个词。"""),
        HumanMessage(content=last_msg),
    ])
    intent = response.content.strip().lower()

    if intent == "rag":
        return {"used_rag": True}
    elif intent == "tool":
        return {"used_tools": True}
    return {}

def rag_node(state: AgentState) -> dict:
    """知识库检索 + 生成回答"""
    kb = KnowledgeBase(settings.database_url)
    last_msg = state["messages"][-1].content
    docs = kb.search(last_msg, k=3)

    context = "\n\n".join([
        f"[来源:{d['source']}]\n{d['content']}"
        for d in docs
    ])

    response = ChatOpenAI(model="gpt-4o").invoke([
        SystemMessage(content=f"{SYSTEM_PROMPT}\n\n知识库:\n{context}"),
        *state["messages"],
    ])

    return {"messages": [response]}

def tool_node(state: AgentState) -> dict:
    """工具调用"""
    llm = ChatOpenAI(model="gpt-4o").bind_tools([
        get_user_orders, get_order_detail, create_ticket
    ])

    response = llm.invoke([
        SystemMessage(content=SYSTEM_PROMPT),
        *state["messages"],
    ])
    return {"messages": [response]}

def chat_node(state: AgentState) -> dict:
    """闲聊"""
    response = ChatOpenAI(model="gpt-4o").invoke([
        SystemMessage(content=SYSTEM_PROMPT),
        *state["messages"],
    ])
    return {"messages": [response]}

def should_route(state: AgentState) -> str:
    """路由到对应节点"""
    last_msg = state["messages"][-1].content
    response = ChatOpenAI(model="gpt-4o-mini").invoke([
        SystemMessage(content="""判断意图:rag / tool / chat"""),
        HumanMessage(content=last_msg),
    ])
    intent = response.content.strip().lower()
    if "rag" in intent: return "rag"
    if "tool" in intent: return "tool"
    return "chat"

# ===== 构建图 =====
builder = StateGraph(AgentState)
builder.add_node("rag", rag_node)
builder.add_node("tool", tool_node)
builder.add_node("tools", ToolNode([get_user_orders, get_order_detail, create_ticket]))
builder.add_node("chat", chat_node)

builder.add_conditional_edges(START, should_route)
builder.add_edge("rag", END)
builder.add_edge("tool", "tools")
builder.add_edge("tools", END)
builder.add_edge("chat", END)

checkpointer = PostgresSaver.from_conn_string(settings.database_url)
graph = builder.compile(checkpointer=checkpointer)

# src/api.py
from fastapi import FastAPI, Depends, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from langchain_core.messages import HumanMessage
from .agent import graph
from .auth import verify_jwt

app = FastAPI()

class ChatRequest(BaseModel):
    message: str
    session_id: str

@app.post("/chat")
async def chat(req: ChatRequest, user_email: str = Depends(verify_jwt)):
    """流式对话接口"""
    config = {"configurable": {"thread_id": req.session_id}}

    async def generate():
        async for event in graph.astream_events(
            {"messages": [HumanMessage(content=req.message)], "user_email": user_email},
            config=config,
            version="v2",
        ):
            kind = event["event"]

            if kind == "on_chat_model_stream":
                content = event["data"]["chunk"].content
                if content:
                    yield f"data: {content}\n\n"

    return StreamingResponse(generate(), media_type="text/event-stream")

@app.post("/upload")
async def upload_doc(file: UploadFile):
    """上传文档到知识库"""
    # 保存文件到 data/docs/
    save_path = f"./data/docs/{file.filename}"
    with open(save_path, "wb") as f:
        f.write(await file.read())

    # 重新构建索引(增量)
    KnowledgeBase(settings.database_url).build_index()

    return {"status": "ok"}

import { ref } from "vue"

const messages = ref([])
const input = ref("")

async function send() {
  if (!input.value) return

  const userMsg = { role: "user", content: input.value }
  messages.value.push({ ...userMsg, id: Date.now() })

  const response = await fetch("/chat", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${localStorage.token}`,
    },
    body: JSON.stringify({
      message: input.value,
      session_id: "user_123",
    }),
  })

  // 流式读取 SSE
  const reader = response.body.getReader()
  const decoder = new TextDecoder()
  const aiMsg = { role: "ai", content: "", id: Date.now() + 1 }
  messages.value.push(aiMsg)

  while (true) {
    const { done, value } = await reader.read()
    if (done) break

    const chunk = decoder.decode(value)
    const lines = chunk.split("\n")
    for (const line of lines) {
      if (line.startsWith("data: ")) {
        aiMsg.content += line.slice(6)
      }
    }
  }

  input.value = ""
}
</script>

🎯 面试常考 ​

Q1：客服 Agent 怎么设计?

核心是「检索 + 工具 + 转人工」:先用 RAG 找到知识库答案,需要订单状态就调工具查,置信度不足或用户不满就转人工。

Q2：怎么控制转人工的时机?

三个信号:检索置信度低于阈值、用户连续两次追问同一问题没解决、用户明确要求人工。转人工时要把上下文一起带过去,不要让用户重说一遍。

Q3：怎么评估客服 Agent?

解决率(无需人工即解决的比例)、转人工准确率、首次响应时间、平均处理时长,再加人工满意度抽样。
