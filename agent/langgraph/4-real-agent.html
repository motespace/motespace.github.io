LangGraph 实战:完整 Agent ​

本章通过两个完整项目,把 LangGraph 的核心能力串起来。 项目 1:ReAct 风格工具调用 Agent;项目 2:Plan-and-Execute Agent。

一、项目 1:ReAct Agent(支持工具调用) ​

python
"""
功能:支持工具调用的 ReAct Agent
- 自动选择工具(get_weather / search_docs / calculate)
- 多步推理(直到 LLM 觉得有足够信息)
- Memory 持久化(支持多轮对话)
- Human-in-the-Loop(敏感操作前确认)
"""
from typing import TypedDict, Annotated, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import ToolNode
from langgraph.checkpoint.memory import MemorySaver
from langgraph.graph.message import add_messages
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, BaseMessage

# ===== 1. 工具 =====
@tool
def get_weather(location: str) -> str:
    """查询指定城市的天气信息,包括温度、天气状况。"""
    # 实际调用天气 API
    return f"{location}:晴,25°C,湿度 60%,风速 3 级"

@tool
def search_docs(query: str, max_results: int = 3) -> str:
    """在技术文档库中搜索。"""
    return f"找到 {max_results} 条关于 '{query}' 的文档..."

@tool
def calculate(expression: str) -> str:
    """计算数学表达式。"""
    try:
        return f"结果: {eval(expression)}"
    except Exception as e:
        return f"错误: {e}"

@tool
def send_email(to: str, subject: str, content: str) -> str:
    """发送邮件(敏感操作,需用户确认)。"""
    # 实际发送邮件前,interrupt 让用户确认
    from langgraph.prebuilt import interrupt
    confirmed = interrupt(f"确认发送邮件给 {to}?主题: {subject}")
    if confirmed == "yes":
        return f"邮件已发送给 {to}"
    return "用户取消"

tools = [get_weather, search_docs, calculate, send_email]

# ===== 2. LLM + 工具绑定 =====
llm = ChatOpenAI(model="gpt-4o").bind_tools(tools)

# ===== 3. State =====
class AgentState(TypedDict):
    messages: Annotated[list[BaseMessage], add_messages]

# ===== 4. 节点 =====
def call_model(state: AgentState):
    """调用 LLM 决策下一步"""
    response = llm.invoke(state["messages"])
    return {"messages": [response]}

# ===== 5. 条件分支 =====
def should_continue(state: AgentState) -> Literal["tools", "__end__"]:
    """决定:执行工具 or 结束"""
    last_msg = state["messages"][-1]
    if last_msg.tool_calls:
        return "tools"
    return END

# ===== 6. 构建图 =====
graph_builder = StateGraph(AgentState)

graph_builder.add_node("agent", call_model)
graph_builder.add_node("tools", ToolNode(tools))

graph_builder.add_edge(START, "agent")
graph_builder.add_conditional_edges("agent", should_continue)
graph_builder.add_edge("tools", "agent")  # 工具执行后回到 agent

memory = MemorySaver()
graph = graph_builder.compile(
    checkpointer=memory,
    interrupt_before=["tools"],  # 工具执行前可以中断
)

# ===== 7. 使用 =====
config = {"configurable": {"thread_id": "user_123"}}

# 第 1 轮
result = graph.invoke(
    {"messages": [HumanMessage(content="北京天气怎么样?")]},
    config=config,
)
print(result["messages"][-1].content)

# 第 2 轮(自动加载历史)
result = graph.invoke(
    {"messages": [HumanMessage(content="那上海呢?")]},
    config=config,
)
print(result["messages"][-1].content)

二、项目 2:Plan-and-Execute Agent ​

python
"""
功能:先 LLM 规划步骤,再逐步执行
- 适合复杂多步任务
- 每步独立执行,失败可重试
- 最终汇总结果
"""
from typing import TypedDict, Annotated, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.types import Command
from langgraph.checkpoint.memory import MemorySaver
from langgraph.graph.message import add_messages
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage
import json

llm = ChatOpenAI(model="gpt-4o")

class PlanExecuteState(TypedDict):
    input: str
    plan: list[dict]
    past_steps: Annotated[list[tuple], lambda old, new: old + [new]]
    response: str
    messages: Annotated[list, add_messages]

# ===== 1. Planner =====
def planner(state: PlanExecuteState) -> Command[Literal["executor"]]:
    """生成执行计划"""
    response = llm.invoke([
        SystemMessage(content="""你是一个任务规划专家。
为用户任务生成执行计划,返回 JSON 数组,每个元素:
{"step": 1, "action": "search_web", "input": "LangChain 2024"}
"""),
        HumanMessage(content=f"任务: {state['input']}"),
    ])

    plan = json.loads(response.content)
    return Command(
        update={"plan": plan},
        goto="executor",
    )

# ===== 2. Executor =====
def executor(state: PlanExecuteState) -> Command[Literal["executor", "replanner"]]:
    """执行当前步骤"""
    plan = state["plan"]
    past_steps = state.get("past_steps", [])

    if not plan:
        return Command(update={}, goto="replanner")

    # 取第一个未执行的步骤
    task = plan[0]
    print(f"执行: {task}")

    # 实际执行(调工具或 LLM)
    response = llm.invoke([
        HumanMessage(content=f"执行步骤 {task['step']}: {task['action']} 输入 {task.get('input', '')}"),
    ])

    # 更新:已执行步骤 + 剩余计划
    return Command(
        update={
            "past_steps": (task, response.content),
            "plan": plan[1:],  # 移除已执行
            "messages": [response],
        },
        goto="executor",  # 继续执行
    )

# ===== 3. Replanner(可选,根据结果调整计划) =====
def replanner(state: PlanExecuteState) -> Command[Literal["executor", END]]:
    """根据执行结果,决定继续或结束"""
    if not state["plan"]:
        # 所有步骤完成,生成最终回答
        response = llm.invoke([
            SystemMessage(content="基于以下执行结果,生成最终回答。"),
            HumanMessage(content=str(state["past_steps"])),
        ])
        return Command(
            update={"response": response.content},
            goto=END,
        )

    # 还有步骤,继续
    return Command(update={}, goto="executor")

# ===== 4. 构建图 =====
builder = StateGraph(PlanExecuteState)
builder.add_node("planner", planner)
builder.add_node("executor", executor)
builder.add_node("replanner", replanner)

builder.add_edge(START, "planner")
builder.add_edge("planner", "executor")
builder.add_edge("executor", "replanner")
builder.add_edge("replanner", END)

memory = MemorySaver()
graph = builder.compile(checkpointer=memory)

# ===== 5. 使用 =====
config = {"configurable": {"thread_id": "session_1"}}

result = graph.invoke(
    {"input": "研究 2024 年最热门的 Python Web 框架,对比 FastAPI 和 Django"},
    config=config,
)

print(result["response"])

三、流式输出(用户体验) ​

python
# 1. 看每个节点的输出
for chunk in graph.stream({"input": "你好"}, config=config):
    print(f"\n[Node Output] {chunk}")

# 2. 流式 events(token 级)
async def stream_agent():
    async for event in graph.astream_events(
        {"input": "北京天气?"},
        config=config,
        version="v2",
    ):
        kind = event["event"]

        if kind == "on_tool_start":
            print(f"\n🔧 工具: {event['name']}")
        elif kind == "on_tool_end":
            print(f"✅ 返回: {event['data'].get('output', '')[:80]}")
        elif kind == "on_chat_model_stream":
            content = event["data"]["chunk"].content
            if content:
                print(content, end="", flush=True)
        elif kind == "on_chain_start" and "name" in event:
            print(f"\n📍 节点: {event['name']}")

asyncio.run(stream_agent())

五、生产部署 ​

python
# FastAPI 包装
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()

class ChatRequest(BaseModel):
    user_id: str
    message: str

class ChatResponse(BaseModel):
    reply: str

@app.post("/chat", response_model=ChatResponse)
async def chat(req: ChatRequest):
    try:
        config = {"configurable": {"thread_id": req.user_id}}
        result = await graph.ainvoke(
            {"messages": [HumanMessage(content=req.message)]},
            config=config,
        )
        return ChatResponse(reply=result["messages"][-1].content)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

# 流式响应(WebSocket)
from fastapi import WebSocket

@app.websocket("/chat/stream")
async def chat_stream(ws: WebSocket):
    await ws.accept()
    config = {"configurable": {"thread_id": ws.query_params.get("user_id")}}

    try:
        while True:
            user_msg = await ws.receive_text()
            async for event in graph.astream_events(
                {"messages": [HumanMessage(content=user_msg)]},
                config=config,
                version="v2",
            ):
                if event["event"] == "on_chat_model_stream":
                    content = event["data"]["chunk"].content
                    if content:
                        await ws.send_json({"chunk": content})
    except Exception:
        await ws.close()

🎯 面试常考 ​

Q1：完整的 Agent 图怎么搭?

至少三节点:agent(调模型)→ tools(执行工具)→ 条件边(有 tool_calls 回到 agent,否则 END),再加 checkpointer 支持恢复。

Q2：怎么防止 Agent 死循环?

限制最大轮数;工具调用连续 N 次无进展就终止;给每个节点设置超时;在提示词里明确终止条件。

Q3：怎么调试 Agent?

用 app.stream(state, stream_mode="values") 把每一步的 State 打出来,能看到模型到底把状态改成了什么;配合 LangSmith 看完整 trace。
