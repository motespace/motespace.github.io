Multi-Agent 多智能体系统 ​

Multi-Agent = 多个 Agent 分工协作,完成复杂任务。 单个 Agent 解决不了的复杂任务(研究 + 写作 + 审阅),用 Multi-Agent 拆分。

一、为什么需要 Multi-Agent? ​

单 Agent 局限:
- 上下文窗口有限(虽然有 200K,但还是有限)
- 一个 Agent 处理多领域,容易混乱
- 错误难以隔离

Multi-Agent 优势:
✅ 分工明确(研究 / 写作 / 审阅各一个 Agent)
✅ 可独立调试
✅ 可扩展(增加 Agent 就行)
✅ 更接近真实团队协作

二、Multi-Agent 三大模式 ​

模式 1:Supervisor(主管模式) ​

          Supervisor(主管)
          /    |    \
         ↓     ↓     ↓
    Researcher  Writer  Reviewer
    (研究)    (写作)  (审阅)

主管决策:
- 用户问题 → 派给 Researcher
- Researcher 完成 → 派给 Writer
- Writer 完成 → 派给 Reviewer
- Reviewer 完成 → 总结输出

模式 2:Swarm(群体模式) ​

无主管,Agent 之间直接通信:

    A ←→ B ←→ C
    ↕    ↕    ↕
    D ←→ E ←→ F

适用:Agent 之间需要频繁交流
例:辩论(Debate)

模式 3:Pipeline(流水线模式) ​

任务像流水线一样按顺序执行:

Input → Agent A → Agent B → Agent C → Output
        研究       写       审

三、Supervisor 模式实战 ​

python
"""
三个 Agent:
- ResearchAgent:搜索资料
- WriterAgent:写文章
- ReviewerAgent:审阅

Supervisor:决定下一步派给谁
"""
from typing import TypedDict, Annotated, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.types import Command
from langgraph.checkpoint.memory import MemorySaver
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage

llm = ChatOpenAI(model="gpt-4o")

# ===== 1. State =====
class TeamState(TypedDict):
    task: str
    research_result: str
    draft: str
    review_result: str
    next_agent: str  # 下一个要执行的 Agent

# ===== 2. 三个 Agent =====
def research_agent(state: TeamState):
    """研究 Agent"""
    response = llm.invoke([
        SystemMessage(content="你是研究助手,负责搜集相关信息。"),
        HumanMessage(content=f"任务:{state['task']}\n\n搜集相关资料,给出 5 个关键事实。"),
    ])
    return {
        "research_result": response.content,
        "next_agent": "writer",
    }

def writer_agent(state: TeamState):
    """写作 Agent"""
    response = llm.invoke([
        SystemMessage(content="你是技术写作助手,基于研究资料写一篇文章。"),
        HumanMessage(content=f"""
任务:{state['task']}
研究资料:{state['research_result']}

请基于以上研究写一篇文章(500 字)。
"""),
    ])
    return {
        "draft": response.content,
        "next_agent": "reviewer",
    }

def reviewer_agent(state: TeamState):
    """审阅 Agent"""
    response = llm.invoke([
        SystemMessage(content="你是审稿人,负责审查文章质量。"),
        HumanMessage(content=f"""
草稿:{state['draft']}

审查要点:
1. 内容准确性
2. 逻辑清晰度
3. 语言表达

如有改进建议,给出;否则说"通过"。
"""),
    ])
    return {
        "review_result": response.content,
        "next_agent": "end",
    }

# ===== 3. Supervisor(路由) =====
def supervisor(state: TeamState) -> Command[Literal["research", "writer", "reviewer", END]]:
    """根据 next_agent 字段路由"""
    next_agent = state["next_agent"]

    if next_agent == "end":
        return Command(update={}, goto=END)

    return Command(update={}, goto=next_agent)

# ===== 4. 构建图 =====
builder = StateGraph(TeamState)

builder.add_node("supervisor", supervisor)
builder.add_node("research", research_agent)
builder.add_node("writer", writer_agent)
builder.add_node("reviewer", reviewer_agent)

builder.add_edge(START, "supervisor")

# supervisor 派发给谁
builder.add_edge("research", "supervisor")
builder.add_edge("writer", "supervisor")
builder.add_edge("reviewer", "supervisor")

memory = MemorySaver()
graph = builder.compile(checkpointer=memory)

# ===== 5. 使用 =====
config = {"configurable": {"thread_id": "task_001"}}

result = graph.invoke(
    {"task": "介绍 LangGraph 的核心概念"},
    config=config,
)

print(result["research_result"])
print("---")
print(result["draft"])
print("---")
print(result["review_result"])

四、动态 Supervisor(LLM 决策) ​

python
def dynamic_supervisor(state: TeamState) -> Command:
    """用 LLM 决定下一步派给谁"""
    response = llm.invoke([
        SystemMessage(content=f"""
你是 Supervisor,管理这些 Agent:
- research: 搜索资料
- writer: 写文章
- reviewer: 审阅

当前状态:
- 研究:{state.get('research_result', '未完成')}
- 草稿:{state.get('draft', '未完成')}
- 审阅:{state.get('review_result', '未完成')}

返回下一步:"research" / "writer" / "reviewer" / "end"
只返回一个词。
"""),
    ])

    next_agent = response.content.strip().lower()
    if next_agent not in ["research", "writer", "reviewer"]:
        next_agent = "end"

    return Command(update={}, goto=next_agent)

五、Swarm 模式(辩论) ​

python
"""
两个 Agent 辩论,达成一致后输出:
- ProAgent:支持观点
- ConAgent:反对观点
- Mediator:判断是否一致
"""
from typing import TypedDict, Annotated, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.types import Command
from langgraph.checkpoint.memory import MemorySaver

class DebateState(TypedDict):
    topic: str
    pro_arguments: list[str]
    con_arguments: list[str]
    round: int
    consensus: bool

def pro_agent(state: DebateState):
    response = llm.invoke([
        SystemMessage(content=f"你支持'{state['topic']}',给出 3 个理由。"),
    ])
    return {"pro_arguments": [response.content]}

def con_agent(state: DebateState):
    response = llm.invoke([
        SystemMessage(content=f"你反对'{state['topic']}',给出 3 个理由。"),
    ])
    return {"con_arguments": [response.content]}

def mediator(state: DebateState) -> Command[Literal["pro", "con", END]]:
    """判断是否达成共识"""
    new_round = state["round"] + 1

    # 超过 3 轮还没共识,强制结束
    if new_round >= 3:
        return Command(update={"round": new_round}, goto=END)

    # 让 LLM 判断
    response = llm.invoke([
        SystemMessage(content=f"""
主题:{state['topic']}
支持方:{state['pro_arguments'][-1] if state['pro_arguments'] else '无'}
反对方:{state['con_arguments'][-1] if state['con_arguments'] else '无'}

是否达成共识?输出"YES"或"NO"。
"""),
    ])

    if "YES" in response.content:
        return Command(update={"consensus": True, "round": new_round}, goto=END)

    return Command(update={"round": new_round}, goto="pro")  # 下一轮

# 构建图(简化)
builder = StateGraph(DebateState)
builder.add_node("pro", pro_agent)
builder.add_node("con", con_agent)
builder.add_node("mediator", mediator)

builder.add_edge(START, "pro")
builder.add_edge("pro", "con")
builder.add_edge("con", "mediator")
builder.add_edge("mediator", "pro")  # 循环

graph = builder.compile()
result = graph.invoke({"topic": "AI 是否会取代程序员", "pro_arguments": [], "con_arguments": [], "round": 0, "consensus": False})
print(result)

from crewai import Agent, Task, Crew

# 1. 定义 Agent
researcher = Agent(
    role="研究员",
    goal="搜集准确信息",
    backstory="你是资深研究员",
    tools=[search_tool],
    llm="gpt-4o",
)

writer = Agent(
    role="作者",
    goal="写高质量文章",
    backstory="你是技术作家",
    llm="gpt-4o",
)

# 2. 定义 Task
research_task = Task(
    description="研究 LangGraph 最新进展",
    agent=researcher,
    expected_output="5 个关键点",
)

write_task = Task(
    description="基于研究写一篇文章",
    agent=writer,
    expected_output="500 字文章",
    context=[research_task],  # 依赖上一任务
)

# 3. 创建 Crew
crew = Crew(
    agents=[researcher, writer],
    tasks=[research_task, write_task],
    verbose=True,
)

result = crew.kickoff()
print(result)

八、何时用哪个模式? ​

Supervisor(主管):
  ✅ 任务链清晰(研究 → 写 → 审)
  ✅ 主管统一调度
  ✅ 最常用

Swarm(群体):
  ✅ Agent 之间需频繁交流
  ✅ 适合辩论 / 协商场景
  ✅ 实现较复杂

Pipeline(流水线):
  ✅ 任务严格按阶段
  ✅ 不需要复杂决策
  最简单

外部框架(AutoGen / CrewAI):
  ✅ 不想自己实现
  ✅ 标准模式
  ✅ 快速原型
