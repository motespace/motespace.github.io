一、Token 基础 ​

什么是 Token? ​

python
import tiktoken

# OpenAI 官方 tokenizer
enc = tiktoken.encoding_for_model("gpt-4")

text = "你好,世界!Hello World"
tokens = enc.encode(text)
print(tokens)
# 中文大约 1 字 = 1-2 token,英文 1 单词 ≈ 1-2 token

# 估算 token 数(常用)
def estimate_tokens(text: str, model: str = "gpt-4") -> int:
    enc = tiktoken.encoding_for_model(model)
    return len(enc.encode(text))

print(estimate_tokens("你好,世界"))  # 4-6

价格计算示例 ​

python
def calculate_cost(
    model: str,
    prompt_tokens: int,
    completion_tokens: int,
) -> float:
    """计算一次调用的费用(美元)"""
    prices = {
        "gpt-4o": (2.5, 10),
        "gpt-4-turbo": (10, 30),
        "gpt-3.5-turbo": (0.5, 1.5),
        "claude-3-5-sonnet": (3, 15),
    }
    input_price, output_price = prices[model]
    cost = (prompt_tokens * input_price + completion_tokens * output_price) / 1_000_000
    return cost

# 例:1k tokens 输入 + 500 tokens 输出,GPT-4
cost = calculate_cost("gpt-4-turbo", 1000, 500)
print(f"费用: ${cost:.4f}")  # $0.025

# 如果每天 100k 次调用,每月成本:
monthly = cost * 100_000 * 30
print(f"每月: ${monthly:.2f}")  # $75,000

1. 模型路由(Model Routing) ​

python
def smart_route(prompt: str, budget: str = "low") -> str:
    """根据任务复杂度选模型"""
    if budget == "low":
        return "gpt-3.5-turbo"  # 便宜
    elif budget == "medium":
        return "gpt-4o"
    else:
        return "gpt-4"  # 最贵但最强

def call_with_route(prompt: str, complexity: str = "low"):
    """简单任务用便宜模型,复杂任务用强模型"""
    if complexity == "simple":
        model = "gpt-3.5-turbo"
    elif complexity == "medium":
        model = "gpt-4o"
    else:
        model = "gpt-4"

    return client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
    )

# 用便宜模型做意图识别 + 强模型做生成
def classify_intent(user_input: str) -> str:
    """用 GPT-3.5 分类"""
    response = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[{
            "role": "user",
            "content": f"判断意图(simple/medium/complex): {user_input}"
        }],
    )
    return response.choices[0].message.content.strip().lower()

def smart_response(user_input: str):
    intent = classify_intent(user_input)
    return call_with_route(user_input, intent)

4. 缓存(避免重复调用) ​

python
import hashlib
from functools import lru_cache

# 用 prompt hash 做缓存 key
cache = {}

def cached_call(prompt: str, model: str = "gpt-3.5-turbo") -> str:
    cache_key = hashlib.md5(f"{model}:{prompt}".encode()).hexdigest()

    if cache_key in cache:
        print("命中缓存")
        return cache[cache_key]

    response = client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
    )
    result = response.choices[0].message.content
    cache[cache_key] = result
    return result

处理 Rate Limit ​

python
import time
from openai import RateLimitError

def call_with_retry(prompt: str, max_retries: int = 3):
    """带指数退避的重试"""
    for attempt in range(max_retries):
        try:
            return client.chat.completions.create(
                model="gpt-4",
                messages=[{"role": "user", "content": prompt}],
            )
        except RateLimitError as e:
            if attempt == max_retries - 1:
                raise
            wait = 2 ** attempt  # 1, 2, 4 秒
            print(f"Rate limited, waiting {wait}s...")
            time.sleep(wait)

# 并发控制(Semaphore)
import asyncio
from openai import AsyncOpenAI

async_client = AsyncOpenAI()
semaphore = asyncio.Semaphore(10)  # 最多 10 并发

async def safe_call(prompt: str):
    async with semaphore:
        return await async_client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
        )

async def batch_call(prompts: list[str]):
    """批量并发调用,但不超过 rate limit"""
    tasks = [safe_call(p) for p in prompts]
    return await asyncio.gather(*tasks, return_exceptions=True)

import logging
from datetime import datetime

# 1. 简单日志
def call_with_logging(prompt: str):
    start = time.perf_counter()
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],

# 2. 完整 cost tracker
class CostTracker:
    def __init__(self, monthly_budget: float):
        self.monthly_budget = monthly_budget
        self.total_cost = 0

    def track(self, cost: float) -> bool:
        """记录花费,返回是否还能继续"""
        self.total_cost += cost
        if self.total_cost > self.monthly_budget * 0.8:
            print(f"⚠️  已花费 80% 月预算: ${self.total_cost:.2f}")
        if self.total_cost > self.monthly_budget:
            print(f"❌ 超出预算! ${self.total_cost:.2f} > ${self.monthly_budget}")
            return False
        return True

tracker = CostTracker(monthly_budget=100)
# ... 每次调用:tracker.track(cost)

🎯 面试常考 ​

Q1：Token 成本怎么算?

总费用 = 输入 token 数 × 输入单价 + 输出 token 数 × 输出单价。输入单价通常远低于输出。

Q2：怎么降低 token 成本?

① 精简 system prompt(去掉没用的示例);② 控制历史对话长度(滑动窗口 / 摘要);③ 缓存固定前缀(部分厂商支持 prompt caching);④ 输出限制 max_tokens;⑤ 路由:简单问题走小模型。

Q3：Rate Limit 是什么?

厂商对每分钟请求数或每分钟 token 数的限制。超了会返回 429。生产环境要自己做并发控制和指数退避重试。
