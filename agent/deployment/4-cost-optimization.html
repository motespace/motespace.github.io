成本优化 ​

AI 应用最大的隐性成本 = Token 费用。 优化得当可降本 50%-80%。

一、成本结构 ​

总成本 = LLM Token 费用 + 基础设施费用 + 运维成本

典型 AI 应用成本占比:
- LLM Token:60-80%  ← 最大开销
- 基础设施:10-30%
- 运维:5-10%

二、LLM Token 优化(8 大策略) ​

1. 模型路由 ​

python
def smart_llm(prompt: str, task_complexity: str) -> str:
    """根据任务复杂度选模型"""
    if task_complexity == "simple":
        # 简单任务用便宜模型
        model = "gpt-3.5-turbo"  # $0.50/1M input
    elif task_complexity == "medium":
        model = "gpt-4o-mini"  # $0.15/1M input
    else:
        model = "gpt-4o"  # $2.50/1M input

    return call_llm(model, prompt)

# 用 GPT-3.5 做意图分类,用 GPT-4o 生成答案
def route_and_answer(user_input: str) -> str:
    # 1. 简单任务用便宜模型
    intent = smart_llm(f"分类: {user_input}", "simple")

    # 2. 根据意图选模型
    if intent in ["chitchat", "faq"]:
        return smart_llm(user_input, "simple")
    else:
        return smart_llm(user_input, "complex")

节省:50-80%

2. Prompt 压缩 ​

python
def compress_prompt(prompt: str) -> str:
    """压缩 prompt,保留关键信息"""
    response = llm.invoke([
        SystemMessage(content="""
你是 Prompt 压缩专家。保留关键信息,去掉冗余,压缩到原长度的 50%。

输入 Prompt:
{prompt}

压缩后(只输出压缩结果):"""),
        HumanMessage(content=prompt),
    ])
    return response.content

# 用
short_prompt = compress_prompt(long_prompt)
response = llm.invoke(short_prompt)

节省:30-50%

3. 语义缓存 ​

python
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache
import hashlib

set_llm_cache(InMemoryCache())

def cached_call(prompt: str) -> str:
    """相同输入返回缓存,无需调 LLM"""
    cache_key = hashlib.md5(prompt.encode()).hexdigest()

    # 检查缓存(自定义实现)
    cached = redis_client.get(cache_key)
    if cached:
        return cached

    response = llm.invoke(prompt)
    redis_client.setex(cache_key, 3600, response.content)
    return response.content

节省:30-60%(取决于重复请求比例)

7. 用更便宜的模型做预处理 ​

python
def preprocess_with_cheap_model(text: str) -> str:
    """用便宜模型做预处理"""
    # 提取关键信息
    return llm.invoke(
        f"提取关键信息:{text}",
        model="gpt-3.5-turbo",
    ).content

def final_answer(text: str) -> str:
    compressed = preprocess_with_cheap_model(text)
    return llm.invoke(
        f"基于: {compressed}",
        model="gpt-4o",
    ).content

节省:40-60%

8. 批处理(高吞吐) ​

python
import asyncio
from openai import AsyncOpenAI

client = AsyncOpenAI()

async def batch_process(prompts: list[str]):
    """并发处理多个请求,提高吞吐量"""
    tasks = [client.chat.completions.create(...) for prompt in prompts]
    return await asyncio.gather(*tasks)

# 100 个 prompt 并发处理,完成时间 ≈ 1 个请求时间

节省:时间(不是钱),降低单用户成本

三、降级策略(主模型挂了用备用) ​

python
def call_llm_with_fallback(prompt: str) -> str:
    """主模型 → 备用模型"""
    try:
        # 1. 主模型(GPT-4o)
        return openai_client.invoke(model="gpt-4o", prompt=prompt)
    except RateLimitError:
        # 2. 备用模型(Claude)
        return anthropic_client.invoke(prompt)
    except Exception:
        # 3. 兜底(简单回答)
        return "抱歉,我现在比较忙,请稍后再试"

# 或用 LangChain 的 with_fallbacks
from langchain_core.runnables import RunnableWithFallbacks

main_chain = prompt | ChatOpenAI(model="gpt-4o") | parser
fallback_chain = prompt | ChatAnthropic(model="claude-3-5-haiku") | parser
robust_chain = main_chain.with_fallbacks([fallback_chain])

四、监控成本(实时) ​

python
from prometheus_client import Counter

token_cost = Counter(
    "token_cost_dollars_total",
    "Total dollar cost",
    ["model"],
)

def track_cost(model: str, prompt_tokens: int, completion_tokens: int):
    """追踪实际花费"""
    prices = {
        "gpt-4o": (2.5, 10),  # per 1M
        "gpt-4o-mini": (0.15, 0.6),
        "gpt-3.5-turbo": (0.5, 1.5),
        "claude-3-5-sonnet": (3, 15),
    }

    input_price, output_price = prices[model]
    cost = (
        prompt_tokens * input_price / 1_000_000
        + completion_tokens * output_price / 1_000_000
    )
    token_cost.labels(model=model).inc(cost)

    # 超过预算告警
    if cost > 1.0:
        alert(f"单次请求 ${cost:.2f},模型 {model}")

五、缓存策略深度 ​

# 1. 精确缓存(完全相同输入)
cache_key = hash(prompt)

# 2. 语义缓存(相似输入也命中)
from sentence_transformers import SentenceTransformer

encoder = SentenceTransformer("all-MiniLM-L6-v2")

def semantic_cache_lookup(prompt: str, threshold: float = 0.85) -> str | None:
    """找相似的缓存"""
    prompt_emb = encoder.encode(prompt)

    # 遍历所有缓存
    for cached_prompt, cached_emb, cached_response in cache.values():
        sim = (prompt_emb @ cached_emb) / (
            np.linalg.norm(prompt_emb) * np.linalg.norm(cached_emb)
        )
        if sim > threshold:
            return cached_response
    return None

六、成本对比实验 ​

python
# 假设 100 万 req/月,平均 prompt 1000 tokens, output 500 tokens

# 全部用 GPT-4o:
cost = 1_000_000 * (1000 * 2.5 / 1e6 + 500 * 10 / 1e6)
print(f"全部 GPT-4o: ${cost:.0f}/月")
# ≈ $7500/月

# 模型路由(80% 简单 → 3.5, 20% 复杂 → 4o):
cost = 800_000 * (1000 * 0.5 / 1e6 + 500 * 1.5 / 1e6) \
      + 200_000 * (1000 * 2.5 / 1e6 + 500 * 10 / 1e6)
print(f"路由后: ${cost:.0f}/月")
# ≈ $2800/月,节省 63%

# 加缓存(50% 命中率):
cost *= 0.5
print(f"+ 缓存: ${cost:.0f}/月")
# ≈ $1400/月,节省 81%
