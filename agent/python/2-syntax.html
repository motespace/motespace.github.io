Python 是动态类型,但有官方类型注解(PEP 484)。

from typing import List, Dict, Tuple, Set, Optional

user: Dict[str, str] = {"name": "Alice"}
point: Tuple[int, int] = (1, 2)
unique_ids: Set[int] = {1, 2, 3}

# Optional[T] = T 或 None
middle_name: Optional[str] = None

# Python 3.9+ 可直接用 list/dict(无需 import)
scores: list[int] = [90, 85, 95]

# 1. dataclass(超常用!AI Agent / Config 必备)

from typing import List

# 2. Pydantic(数据验证,LangChain / FastAPI 必备)
from pydantic import BaseModel, Field, validator

class User(BaseModel):
    name: str = Field(..., min_length=1, max_length=50)
    age: int = Field(..., ge=0, le=150)
    email: str

    @validator("email")
    def validate_email(cls, v: str) -> str:
        if "@" not in v:
            raise ValueError("invalid email")
        return v.lower()

user = User(name="Alice", age=25, email="ALICE@EXAMPLE.COM")
print(user.email)  # alice@example.com(自动小写)

# 3. 继承 + 抽象类
from abc import ABC, abstractmethod

class BaseAgent(ABC):
    @abstractmethod
    def run(self, input: str) -> str:
        pass

class MyAgent(BaseAgent):
    def run(self, input: str) -> str:
        return f"Processed: {input}"

agent = MyAgent()
print(agent.run("hello"))  # Processed: hello

四、装饰器(AI 框架核心) ​

装饰器 = 不修改原函数,给函数"附加功能"。LangChain / LangGraph 大量使用。

python
# 1. 自定义装饰器
import functools
import time

def timer(func):
    """计时装饰器"""
    @functools.wraps(func)  # 保留原函数元信息
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.3f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "done"

slow_function()  # slow_function took 1.001s

# 2. 带参数的装饰器(配置 retry / log level)
def retry(max_attempts: int = 3):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"Retry {attempt + 1}/{max_attempts}: {e}")
        return wrapper
    return decorator

@retry(max_attempts=3)
def call_llm(prompt: str) -> str:
    """调用 LLM,失败自动重试"""
    # ...
    raise Exception("API timeout")

# 3. 类装饰器(更强大)
class CountCalls:
    def __init__(self, func):
        self.func = func
        self.count = 0

    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"{self.func.__name__} called {self.count} times")
        return self.func(*args, **kwargs)

@CountCalls
def hello():
    print("Hello!")

hello()  # hello called 1 times
hello()  # hello called 2 times

# 2. 自定义上下文管理器
from contextlib import contextmanager

@contextmanager
def timer_block(label: str):
    start = time.perf_counter()
    print(f"[{label}] start")
    yield  # 这里是 with 块的内容
    elapsed = time.perf_counter() - start
    print(f"[{label}] done in {elapsed:.3f}s")

with timer_block("Load data"):
    time.sleep(1)
    # 加载数据...
# [Load data] start
# [Load data] done in 1.001s

# 3. LLM API 调用经常用(限制 budget)
from contextlib import contextmanager

class BudgetExceeded(Exception):
    pass

@contextmanager
def budget_limit(max_cost: float):
    spent = 0
    try:
        yield lambda cost: spent + cost <= max_cost
    finally:
        print(f"Total cost: ${spent:.4f}")

with budget_limit(max_cost=1.0) as can_spend:
    if can_spend(0.5):

七、生成器(处理大数据流) ​

python
# 1. 普通生成器(节省内存)
def read_large_file(path: str):
    with open(path) as f:
        for line in f:
            yield line.strip()

for line in read_large_file("huge.log"):
    print(line)
# 不会一次性读入整个文件

# 2. 异步生成器(AI 流式输出)
async def stream_llm_response(prompt: str):
    """模拟 LLM 流式响应"""
    chunks = ["Hello", " ", "world", "!"]
    for chunk in chunks:
        await asyncio.sleep(0.1)
        yield chunk

async def main():
    async for chunk in stream_llm_response("Hi"):
        print(chunk, end="", flush=True)
    print()

asyncio.run(main())
# Hello world!

🎯 面试常考 ​

Q1：类型提示有什么用?

让 IDE 能补全、让运行时校验、也让 LangChain 能推断链的输入输出类型。AI 开发里几乎所有公开 API 都带类型标注。

Q2：dataclass 有什么价值?

一行声明就得到带 __init__、__repr__、比较运算符的类,不用手写样板代码;配合 frozen=True 还能当不可变配置对象。

Q3：装饰器怎么理解?

接收函数返回新函数的函数,用来在不改动原函数的前提下加功能(日志、缓存、重试)。Python 里的 @property、@dataclass 都是装饰器。
