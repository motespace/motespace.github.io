Python 异步编程(asyncio / await) ​

AI Agent 高并发场景必备。并发调用多个 LLM API / 异步 I/O 操作, 不用异步就只能串行,慢 10 倍不止。

一、为什么需要异步? ​

python
# 串行:慢
import time

def fetch_one(url: str) -> str:
    time.sleep(1)  # 模拟 IO
    return f"data from {url}"

start = time.perf_counter()
results = [fetch_one(f"url_{i}") for i in range(10)]
print(f"Sync: {time.perf_counter() - start:.2f}s")  # ~10s

# 异步:快
import asyncio

async def fetch_one_async(url: str) -> str:
    await asyncio.sleep(1)  # 模拟 IO(可被打断)
    return f"data from {url}"

async def main():
    start = time.perf_counter()
    results = await asyncio.gather(*[fetch_one_async(f"url_{i}") for i in range(10)])
    print(f"Async: {time.perf_counter() - start:.2f}s")  # ~1s

asyncio.run(main())

核心:IO 操作时,异步可以让 CPU 去做别的事,10 个并发请求 ≈ 1 个请求的时间。

二、async / await 基础 ​

python
import asyncio

# async def 定义协程函数
async def hello():
    print("Hello")
    await asyncio.sleep(1)  # await 让出控制权
    print("World")

# 调用方式(必须用 await 或 asyncio.run)
asyncio.run(hello())
# 或在另一个 async 函数中:
async def main():
    await hello()

asyncio.run(main())

三、并发原语(asyncio.gather / Task) ​

python
import asyncio

async def fetch(url: str, delay: float) -> str:
    await asyncio.sleep(delay)
    return f"data from {url}"

# 1. gather:等待所有任务完成
async def main_gather():
    results = await asyncio.gather(
        fetch("url1", 1),
        fetch("url2", 2),
        fetch("url3", 0.5),
    )
    print(results)  # ['data from url1', 'data from url2', 'data from url3']

# 2. create_task:创建后台任务,稍后 await
async def main_task():
    task1 = asyncio.create_task(fetch("url1", 2))
    task2 = asyncio.create_task(fetch("url2", 1))

    # ... 做点别的事 ...
    print("doing other things")

    # 等待结果
    r1 = await task1
    r2 = await task2
    print(r1, r2)

# 3. as_completed:逐个处理完成的(适合流式响应)
async def main_as_completed():
    tasks = [fetch(f"url_{i}", i * 0.5) for i in range(5)]
    for coro in asyncio.as_completed(tasks):
        result = await coro
        print(f"got: {result}")

四、AI 实战:并发调用 LLM API ​

python
import asyncio
from openai import AsyncOpenAI

client = AsyncOpenAI()

async def call_llm(prompt: str, model: str = "gpt-4") -> str:
    """异步调用 LLM"""
    response = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
    )
    return response.choices[0].message.content

async def main():
    # ❌ 串行(慢)
    prompts = ["讲个笑话", "写首诗", "解释量子力学"]
    results = []
    for p in prompts:
        r = await call_llm(p)
        results.append(r)

    # ✅ 并发(快 3 倍)
    tasks = [call_llm(p) for p in prompts]
    results = await asyncio.gather(*tasks)

    for prompt, result in zip(prompts, results):
        print(f"Q: {prompt}\nA: {result}\n")

asyncio.run(main())

五、流式输出(Streaming) ​

python
from openai import AsyncOpenAI

client = AsyncOpenAI()

async def stream_llm(prompt: str):
    """流式调用 LLM,实时显示输出"""
    print("AI: ", end="", flush=True)
    stream = await client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        stream=True,
    )
    async for chunk in stream:
        content = chunk.choices[0].delta.content
        if content:
            print(content, end="", flush=True)
    print()

asyncio.run(stream_llm("用一句话介绍 Python"))

六、LangChain 异步用法 ​

python
# LangChain v0.1+ 全面支持异步
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

llm = ChatOpenAI(model="gpt-4")

# 1. 单次调用
async def chat_once():
    response = await llm.ainvoke([HumanMessage(content="你好")])
    print(response.content)

# 2. 批量并发
async def chat_batch():
    messages_list = [
        [HumanMessage(content="1+1=?")],
        [HumanMessage(content="2+2=?")],
        [HumanMessage(content="3+3=?")],
    ]
    responses = await llm.abatch(messages_list)
    for r in responses:
        print(r.content)

# 3. 流式
async def chat_stream():
    async for chunk in llm.astream([HumanMessage(content="写首诗")]):
        print(chunk.content, end="", flush=True)
    print()

asyncio.run(chat_once())
asyncio.run(chat_batch())
asyncio.run(chat_stream())

七、并发控制(Semaphore) ​

python
import asyncio

# 用 Semaphore 限制并发数(避免 API rate limit)
async def call_with_limit(semaphore: asyncio.Semaphore, prompt: str) -> str:
    async with semaphore:
        # 同时最多 5 个并发
        return await call_llm(prompt)

async def main():
    semaphore = asyncio.Semaphore(5)
    tasks = [call_with_limit(semaphore, p) for p in range(100)]
    results = await asyncio.gather(*tasks)
    # 100 个请求,但始终只有 5 个并发

asyncio.run(main())

八、错误处理 + 超时 ​

python
import asyncio

async def safe_call(prompt: str, timeout: float = 10.0) -> str:
    """带超时和重试的 LLM 调用"""
    for attempt in range(3):
        try:
            return await asyncio.wait_for(call_llm(prompt), timeout=timeout)
        except asyncio.TimeoutError:
            print(f"Attempt {attempt + 1} timed out")
            if attempt == 2:
                return "LLM timeout"
            await asyncio.sleep(2 ** attempt)  # 指数退避
        except Exception as e:
            print(f"Attempt {attempt + 1} failed: {e}")
            if attempt == 2:
                raise

async def main():
    results = await asyncio.gather(
        safe_call("prompt1"),
        safe_call("prompt2"),
        safe_call("prompt3"),
        return_exceptions=True,  # 一个失败不影响其他
    )

asyncio.run(main())

九、实战:Agent 异步工作流 ​

python
import asyncio
from dataclasses import dataclass

@dataclass
class AgentTask:
    name: str
    prompt: str

async def run_agent(task: AgentTask) -> str:
    """执行单个 Agent 任务"""
    print(f"[{task.name}] starting...")
    result = await call_llm(task.prompt)
    print(f"[{task.name}] done")
    return result

async def parallel_agents():
    """并发运行多个 Agent"""
    tasks = [
        AgentTask("Researcher", "调研最新的 LLM 论文"),
        AgentTask("Writer", "基于研究写一篇博客"),
        AgentTask("Reviewer", "审查博客质量"),
    ]

    # 阶段 1:并发研究
    research_results = await asyncio.gather(
        run_agent(tasks[0]),
        run_agent(tasks[0]),  # 模拟并行研究
    )

    # 阶段 2:基于研究写
    blog = await run_agent(AgentTask("Writer", f"基于: {research_results}"))

    # 阶段 3:审查
    review = await run_agent(AgentTask("Reviewer", f"审查: {blog}"))

    return review

asyncio.run(parallel_agents())

十、避坑指南 ​

python
# ❌ 错误 1:在 sync 函数中调用 async(直接 await)
def sync_func():
    await call_llm("p")  # SyntaxError: can't use await outside async

# ✅ 正确:用 asyncio.run
def sync_func():
    asyncio.run(call_llm("p"))

# ❌ 错误 2:忘记 await(只是创建协程,不执行)
async def main():
    call_llm("p")  # 没执行!
    await asyncio.sleep(1)

# ✅ 正确
async def main():
    await call_llm("p")

# ❌ 错误 3:在 async 中用 requests(阻塞 event loop)
import requests  # 同步 HTTP 库
async def main():
    requests.get("url")  # 阻塞!

# ✅ 正确:用 httpx.AsyncClient
import httpx
async def main():
    async with httpx.AsyncClient() as client:
        await client.get("url")

🎯 面试常考 ​

Q1：async/await 解决什么问题?

在 I/O 等待期间让出控制权,单线程就能并发处理大量 I/O。调 LLM API 全是网络 I/O,是 async 的典型场景。

Q2：什么时候不该用 async?

CPU 密集任务(async 不会让 CPU 并行,反而增加开销);或者代码只有一两个串行调用,直接用同步更清晰。

Q3：并发调用多个 LLM 怎么写?

asyncio.gather(*[client.ainvoke(p) for p in prompts]) 或 asyncio.as_completed(...);配合 httpx.AsyncClient 连接复用。
