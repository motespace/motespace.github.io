10.4.1 buildId 注入（DefinePlugin） ​

ts
webpack(config, { dev, isServer, buildId }) {
  if (!isServer && !dev) {
    config.plugins.push(
      new webpack.DefinePlugin({
        // 编译时替换：客户端运行时 process.env.NEXT_PUBLIC_BUILD_ID === buildId 字符串
        'process.env.NEXT_PUBLIC_BUILD_ID': JSON.stringify(buildId),
        'globalThis.__BUILD_ID__': JSON.stringify(buildId),
      }),
    )
  }
}

DefinePlugin 在编译时将所有 process.env.NEXT_PUBLIC_BUILD_ID 替换为字面量值（如 "rMG10zNnVOgx4YNQPfdZ5"），客户端运行时无需查询 process.env，直接拿到字符串。

10.4.2 manifestConfig() — 运行时缓存策略 ​

ts
function manifestConfig() {
  return {
    swDest: '../public/sw-h5.js',   // 输出路径
    clientsClaim: true,              // SW 激活后立即接管页面
    skipWaiting: true,               // 新 SW 安装后跳过 waiting 状态
    cleanupOutdatedCaches: true,     // 删除旧版本缓存
    inlineWorkboxRuntime: true,      // workbox runtime 内联到 sw.js（减少请求数）
    exclude: [/\.*/],               // 不预缓存任何文件（全部走 runtimeCaching）

    // manifest 路径重写：_next/ 替换为 sw-h5/
    manifestTransforms: [
      (manifestEntries) => {
        const manifest = manifestEntries.map((entry) => ({
          ...entry,
          url: entry.url.replace('_next/', 'sw-h5/').replace('//static/', '/static/'),
        }))
        return { manifest, warnings: [] }
      },
    ],

    // 运行时缓存策略
    runtimeCaching: [
      // 1. Next.js 核心 chunk → NetworkFirst，1s 超时
      {
        urlPattern: /\/_next\/static\/chunks\/(runtime|main|react|framework).*\.js$/,
        handler: 'NetworkFirst',
        options: { cacheName: 'h5-next-core', networkTimeoutSeconds: 1 },
      },

      // 2. Next.js 静态资源 → CacheFirst
      {
        urlPattern: /\/_next\/static\//,
        handler: 'CacheFirst',
        options: {
          cacheName: 'h5-next-static',
          expiration: { maxEntries: 100 },
          cacheableResponse: { statuses: [0, 200] },
        },
      },

      // 3. Next.js 图片优化接口 → StaleWhileRevalidate，7 天
      {
        urlPattern: /\/_next\/image\?url=.+/,
        handler: 'StaleWhileRevalidate',
        options: {
          cacheName: 'h5-next-image',
          expiration: { maxEntries: 100, maxAgeSeconds: 7 * 24 * 60 * 60 },
          cacheableResponse: { statuses: [0, 200] },
        },
      },

      // 4. SSR JSON 数据（英文）→ StaleWhileRevalidate，90 天
      {
        urlPattern: /\/_next\/data\/[^/]+\/en\/.+\.json/,
        handler: 'StaleWhileRevalidate',
        options: {
          cacheName: 'h5-json-cache-en',
          expiration: { maxEntries: 100, maxAgeSeconds: 24 * 60 * 60 * 90 },
          cacheableResponse: { statuses: [0, 200] },
        },
      },

      // 5. SSR JSON 数据（中文）
      {
        urlPattern: /\/_next\/data\/[^/]+\/zh\/.+\.json/,
        handler: 'StaleWhileRevalidate',
        options: {
          cacheName: 'h5-json-cache-zh',
          expiration: { maxEntries: 100, maxAgeSeconds: 24 * 60 * 60 * 90 },
          cacheableResponse: { statuses: [0, 200] },
        },
      },

      // 6. 页面导航请求 → NetworkFirst，3s 超时
      {
        urlPattern: ({ request }) => request.mode === 'navigate',
        handler: 'NetworkFirst',
        options: {
          cacheName: 'h5-pages-cache',
          networkTimeoutSeconds: 3,
          expiration: { maxEntries: 100 },
          cacheableResponse: { statuses: [0, 200] },
        },
      },

      // 7. 第三方域名 → NetworkOnly（不缓存）
      {
        urlPattern: ({ url }) =>
          url.hostname === 'ui.invisiblesport.com' ||
          url.hostname.endsWith('.sptpub.com') ||
          url.hostname === 'sptpub.com',
        handler: 'NetworkOnly',
      },
    ],
  }
}

10.5 Service Worker 消息通道 ​

页面与 SW 通过 postMessage 通信。SW 在末尾监听 4 种消息类型：

10.5.1 注入消息监听代码（构建后追加） ​

在 compiler.hooks.done.tap 回调里，将消息通道代码追加到 sw-h5.js 末尾：

ts
compiler.hooks.done.tap('InjectSWInstallHook', () => {
  const swPath = path.join(__dirname, './public/sw-h5.js')
  const injectCode = `
self.addEventListener('message', (event) => {
  if (!event.data) return;

  if (event.data.type === 'PRECACHE_URLS') {
    const { payload, locale } = event.data;
    const cacheName = 'h5-json-cache-' + locale;
    event.waitUntil(
      caches.open(cacheName).then(async (cache) => {
        for (const url of payload ?? []) {
          const cached = await cache.match(url);
          if (cached) continue;
          try {
            const res = await fetch(url, { cache: 'no-store' });
            if (res.ok) await cache.put(url, res.clone());
          } catch (err) {
            console.error('动态预缓存失败', url, err);
          }
        }
      }),
    );
  }

  if (event.data.type === 'CLEAR_JSON_CACHE') {
    const { locale } = event.data;
    event.waitUntil(caches.delete('h5-json-cache-' + locale));
  }

  if (event.data.type === 'RESET_AND_PRECACHE') {
    const { payload, locale } = event.data;
    const cacheName = 'h5-json-cache-' + locale;
    const CONCURRENCY = 5;
    event.waitUntil(
      caches.delete(cacheName)
        .then(() => caches.open(cacheName))
        .then(async (cache) => {
          for (let i = 0; i < payload.length; i += CONCURRENCY) {
            const chunk = payload.slice(i, i + CONCURRENCY);
            await Promise.all(
              chunk.map(async (url) => {
                try {
                  const res = await fetch(url, { cache: 'reload' });
                  if (res.ok) await cache.put(url, res.clone());
                } catch (err) {
                  console.error('precache fail', url);
                }
              }),
            );
          }
        }),
    );
  }

  if (event.data.type === 'SKIP_WAITING') self.skipWaiting();
});
`;
  fs.appendFileSync(swPath, injectCode, 'utf-8');
});

10.5.2 消息类型说明 ​

消息类型触发时机作用
PRECACHE_URLS追加预缓存时追加新 URL 到缓存，不删旧数据
CLEAR_JSON_CACHE切换语言时删除旧语言的所有 JSON 缓存
RESET_AND_PRECACHESW 注册时 / Build ID 变化时删除旧缓存，重新批量写入，并发数 5
SKIP_WAITING新 SW 安装完成后通知新 SW 跳过 waiting，立即激活

10.6.2 注销旧 SW（防止多版本冲突） ​

ts
async function unregisterForeignServiceWorkers() {
  if (!('serviceWorker' in navigator) || process.env.NODE_ENV !== 'production') return

  const registrations = await navigator.serviceWorker.getRegistrations()
  for (const registration of registrations) {
    const url = registration.active?.scriptURL ||
                registration.installing?.scriptURL ||
                registration.waiting?.scriptURL

    // 当前版本 SW，保留
    if (
      url?.includes('/sw-h5.js') &&
      process.env.NEXT_PUBLIC_BUILD_ID === getItem(StorageEnum.BuildId)
    ) {
      continue
    }

    await registration.unregister()  // 旧版本 SW 注销
  }

  // 清理所有旧缓存
  const cacheNames = await caches.keys()
  await Promise.all(cacheNames.map((name) => caches.delete(name)))
}

10.6.3 注册 SW + 监听更新 ​

ts
navigator.serviceWorker
  .register(`/sw-h5.js`, {
    updateViaCache: 'none',  // SW 本身不从缓存更新
  })
  .then((registration) => {
    registration.addEventListener('updatefound', () => {
      const newWorker = registration.installing
      if (newWorker) {
        newWorker.addEventListener('statechange', () => {
          if (newWorker.state === 'installed') {
            newWorker.postMessage({ type: 'SKIP_WAITING' })
            requestPrecache(newWorker)
          }
        })
      }
    })
  })
  .then(() => navigator.serviceWorker.ready)
  .then(async () => {
    requestPrecache(navigator.serviceWorker.controller)
  })

10.6.4 构建预缓存 payload ​

ts
function buildPrecachePayload(menuInfo, navList, locale, buildId) {
  const payload: string[] = []
  for (const item of menuInfo) {
    const content = jsonToObj(item.content || item?.jumpContentDTO?.content)
    if (!content?.jumpAddress) continue

    const dataUrl = `/_next/data/${buildId}/${locale}${content.jumpAddress}.json`

    // 娱乐场/体育详情页：需要额外缓存 tab 参数对应的 JSON
    if (
      content.jumpAddress.includes('/casino') ||
      content.jumpAddress.includes('/sport')
    ) {
      const menuId = item?.id
      if (menuId) {
        try {
          const tabsMenu = await homeIns.getSecondLevelMenuUsingPost({ menuId })
          const firstTabId = tabsMenu?.[0]?.menuId
          if (firstTabId) payload.push(`${dataUrl}?tab=${firstTabId}`)
        } catch {}
      }
    }
    payload.push(dataUrl)
  }
  return Array.from(new Set(payload))
}

10.6.5 requestPrecache（带版本校验） ​

ts
async function requestPrecache(worker: ServiceWorker | null | undefined) {
  if (!worker) return

  const cacheBuildId = getItem(StorageEnum.BuildId)  // localStorage 中存的上次 buildId

  // Build ID 没变，且缓存已有数据 → 完全跳过
  if (cacheBuildId === buildId) {
    try {
      const cache = await caches.open(`h5-json-cache-${locale}`)
      if ((await cache.keys()).length > 0) return
    } catch {}
  }

  precacheRequested = true
  setItem(StorageEnum.BuildId, buildId, StorageTypeEnum.LocalStorage)

  worker.postMessage({
    type: 'RESET_AND_PRECACHE',
    payload: uniquePayload,
    locale,
  })
}

10.7 版本管理与缓存失效 ​

10.7.1 版本管理三层机制 ​

层次机制作用
Build IDNext.js 构建随机生成，每次 build 必变JSON URL 路径包含 buildId，路径变了 → 缓存自动失效
StorageEnum.BuildIdlocalStorage 存储当前 buildId页面加载时对比，判断是否需要重建缓存
skipWaiting + clients.claimSW 配置项新 SW 激活后立即接管页面，无需用户刷新

10.7.2 版本更新完整流程 ​

开发者执行 next build
    │
    ▼
Next.js 生成新 buildId = "xyz789..."
    │
    ▼
webpack DefinePlugin 注入新 buildId 到客户端 bundle
    │
    ▼
Deploy，用户访问页面
    │
    ▼
_app.tsx: unregisterForeignServiceWorkers()
    ├─ 发现旧 SW Build ID 不一致 → 注销旧 SW
    └─ 删除所有旧缓存
    │
    ▼
requestPrecache()
    ├─ cacheBuildId !== buildId（新版本）
    ├─ setItem(StorageEnum.BuildId, 新buildId)
    └─ postMessage({ type: 'RESET_AND_PRECACHE' })
            │
            ▼
    SW: caches.delete() → caches.open() → 并发 5 个请求重新写入

10.7.3 第二次访问为何不重新预缓存 ​

首次访问（Build ID: abc123）
    ├─ requestPrecache → SW 缓存 JSON 数据
    └─ localStorage: BuildId = abc123

第二次访问（Build ID: abc123，不变）
    ├─ cacheBuildId === buildId ✓
    ├─ cache.match() 有数据
    └─ return（完全跳过，无网络请求，秒开）

10.8 运行时缓存策略详解 ​

10.8.1 策略类型 ​

策略行为适用场景
CacheFirst先查缓存，命中直接返回；未命中请求网络并写入缓存CSS/JS/图片等静态资源
NetworkFirst先请求网络，失败时回退到缓存；可设 networkTimeoutSeconds核心 JS chunk、页面导航
StaleWhileRevalidate立即返回缓存，后台异步请求网络更新缓存SSR JSON 数据，优先展示可缓存内容
NetworkOnly不缓存，直接网络请求第三方 API / 登录接口

10.8.2 请求匹配流程 ​

浏览器发起请求
    │
    ├─ h5-next-core（核心 chunk）→ NetworkFirst，1s 超时
    ├─ h5-next-static（静态资源）→ CacheFirst，maxEntries: 100
    ├─ h5-next-image（图片）→ StaleWhileRevalidate，7 天过期
    ├─ h5-json-cache-en/zh（SSR JSON）→ StaleWhileRevalidate，90 天过期
    ├─ h5-pages-cache（navigate）→ NetworkFirst，3s 超时
    └─ 第三方域名 → NetworkOnly

10.9 usePWAStatus 安装检测 ​

路径：src/hooks/usePWAStatus/index.ts

10.9.1 安装状态检测 ​

ts
interface PWAStatus {
  isStandalone: boolean    // 是否以 PWA 独立模式运行
  isInstalled: boolean     // 是否已安装到主屏
  platform: string         // android / ios / windows / macos / desktop
  installationMethod: string  // browser / share-sheet / twa / manual-menu / auto-prompt
  displayMode?: string    // standalone / fullscreen / minimal-ui / browser
}

export function usePWAStatus() {
  const [pwaStatus, setPwaStatus] = useState<PWAStatus>({...})
  const [canShowInstallPrompt, setCanShowInstallPrompt] = useState(false)
  const [deferredPrompt, setDeferredPrompt] = useState<any>(null)

  useEffect(() => {
    // 1. 检测 PWA 独立模式
    const isStandalone =
      window.matchMedia('(display-mode: standalone)').matches ||
      (navigator as any).standalone === true

    const platform = detectPlatform()
    const installationMethod = detectInstallationMethod(isStandalone, platform)

    setPwaStatus({ isStandalone, isInstalled: isStandalone, platform, installationMethod })

    // 2. 监听安装提示（Android Chrome）
    window.addEventListener('beforeinstallprompt', (e) => {
      e.preventDefault()
      setDeferredPrompt(e)
      setCanShowInstallPrompt(true)
    })

    // 3. 监听安装完成
    window.addEventListener('appinstalled', () => {
      setPwaStatus((prev) => ({ ...prev, isInstalled: true, isStandalone: true }))
    })

    // 4. 监听显示模式变化
    const mediaQuery = window.matchMedia('(display-mode: standalone)')
    mediaQuery.addEventListener('change', (e) => {
      setPwaStatus((prev) => ({ ...prev, isStandalone: e.matches, isInstalled: e.matches }))
    })
  }, [])

  const installPWA = async () => {
    if (!deferredPrompt) return
    deferredPrompt.prompt()
    const { outcome } = await deferredPrompt.userChoice
    if (outcome === 'accepted') setCanShowInstallPrompt(false)
    setDeferredPrompt(null)
  }

  return { pwaStatus, installPWA, canShowInstallPrompt }
}

10.9.2 平台与安装方式检测 ​

ts
function detectPlatform(): string {
  const ua = navigator.userAgent.toLowerCase()
  if (/android/.test(ua)) return 'android'
  if (/iphone|ipad|ipod/.test(ua)) return 'ios'
  if (/win/.test(ua)) return 'windows'
  if (/mac/.test(ua)) return 'macos'
  if (/linux/.test(ua)) return 'linux'
  return 'desktop'
}

function detectInstallationMethod(isStandalone, platform): string {
  if (!isStandalone) return 'browser'
  if (platform === 'ios') return 'share-sheet'      // iOS 通过分享菜单安装
  if (platform === 'android') {
    const isTWA = (navigator as any).standalone === undefined
    return isTWA ? 'twa' : 'manual-menu'
  }
  return 'manual-menu'
}

🎯 面试常考 ​

Q1：Service Worker 是什么?

运行在独立线程的 JS 代理,拦截页面的网络请求,可以缓存资源、离线可用、后台同步。

Q2：生命周期?

Parsed → Installing → Installed(Waiting)→ Activating → Activated。旧 SW 在 Waiting 状态直到所有页面关闭或调用 skipWaiting。

Q3：缓存策略怎么选?

Cache First(静态资源)、Network First(实时数据)、Stale While Revalidate(先给缓存再后台更新)。按资源类型选,别一刀切。

📝 小结 ​

  • SW 是独立线程的代理,拦截请求,能预缓存实现离线。
  • 两种缓存时机:预缓存(build 时生成清单,install 时下载)vs 运行时缓存(首次请求才缓存)。
  • 策略:CacheFirst(静态资源)/ NetworkFirst(接口)/ StaleWhileRevalidate(中间态)/ NetworkOnly(绝不缓存)。
  • 版本更新靠 sw.js 内容变化 → install → waiting → activate 里清旧 cache;生产上做「提示用户刷新」而不是静默接管。
