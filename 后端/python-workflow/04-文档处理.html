文档处理 ​

用 Python 自动生成 / 解析 Markdown / Word / PDF。

一、Markdown 生成 ​

python
# 直接字符串拼
md = '# 报告\\n\\n'
md += '## 数据\\n'
md += '| 日期 | GMV |\\n'
md += '| --- | --- |\\n'
md += '| 2025-01-15 | 128500 |\\n'

with open('report.md', 'w', encoding='utf-8') as f:
    f.write(md)

更结构化的方案:

python
from dataclasses import dataclass

@dataclass
class Report:
    title: str
    sections: list

def render(report):
    out = f'# {report.title}\\n\\n'
    for sec in report.sections:
        out += f'## {sec.title}\\n\\n{sec.body}\\n'
    return out

r = Report(
    title='销售日报',
    sections=[
        {'title': 'GMV', 'body': '128,500 元'},
        {'title': '订单数', 'body': '342 单'},
    ]
)
with open('report.md', 'w', encoding='utf-8') as f:
    f.write(render(r))

二、Word 文档(python-docx) ​

python
from docx import Document
from docx.shared import Pt, Cm
from docx.enum.text import WD_ALIGN_PARAGRAPH

doc = Document()

# 标题
h = doc.add_heading('销售日报 - 2025-01-15', level=1)

# 正文
p = doc.add_paragraph('今日 GMV 128,500 元,较昨日增长 12%。')

# 表格
table = doc.add_table(rows=1, cols=3)
hdr = table.rows[0].cells
hdr[0].text = '商品'
hdr[1].text = '销售额'
hdr[2].text = '订单数'

for row_data in [
    ('iPhone 15', '8999', '12'),
    ('AirPods', '1299', '28'),
]:
    row = table.add_row().cells
    for i, val in enumerate(row_data):
        row[i].text = val

# 图片
doc.add_picture('chart.png', width=Cm(15))

# 字体 / 样式
for run in p.runs:
    run.font.size = Pt(12)

doc.save('report.docx')

三、PDF 生成(reportlab) ​

python
from reportlab.lib.pagesizes import A4
from reportlab.pdfgen import canvas
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont

# 注册中文字体(防乱码)
pdfmetrics.registerFont(TTFont('msyh', 'msyh.ttf'))

c = canvas.Canvas('report.pdf', pagesize=A4)
width, height = A4

c.setFont('msyh', 16)
c.drawString(100, height - 80, '销售日报')

c.setFont('msyh', 12)
c.drawString(100, height - 110, '日期:2025-01-15')
c.drawString(100, height - 130, 'GMV:128,500 元')

# 表格
from reportlab.platypus import SimpleDocTemplate, Table, TableStyle
from reportlab.lib import colors

doc = SimpleDocTemplate('table.pdf', pagesize=A4)
elements = []
t = Table([
    ['商品', '销售额', '订单数'],
    ['iPhone', '8999', '12'],
    ['AirPods', '1299', '28'],
])
t.setStyle(TableStyle([
    ('FONTNAME', (0, 0), (-1, -1), 'msyh'),
    ('BACKGROUND', (0, 0), (-1, 0), colors.grey),
    ('GRID', (0, 0), (-1, -1), 1, colors.black),
]))
elements.append(t)
doc.build(elements)

四、PDF 解析(pdfplumber) ​

python
import pdfplumber

with pdfplumber.open('invoice.pdf') as pdf:
    for page in pdf.pages:
        text = page.extract_text()
        print(text)

        # 提取表格
        tables = page.extract_tables()
        for table in tables:
            for row in table:
                print(row)

        # 提取图片
        for img in page.images:
            print(img['name'], img['x0'])

六、踩坑总结 ​

坑解决
PDF 中文乱码注册中文字体(msyh / 思源黑体)
Word 表格对齐cell.paragraphs[0].alignment
Markdown 渲染表格用 | --- | 或直接 HTML
大 PDF 解析慢分页处理,或转图片 OCR
docx 模板套打用 docx-mailmerge 或 jinja2 + docxtpl
