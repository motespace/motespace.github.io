一、目录监听(watchdog) ​

python
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
import time

class AutoArchiveHandler(FileSystemEventHandler):
    def on_created(self, event):
        if event.is_directory:
            return
        file_path = event.src_path
        if file_path.endswith('.tmp'):
            return  # 忽略临时文件
        print(f'新文件: {file_path}')
        # 触发处理逻辑(上传 / 归档 / 通知)

handler = AutoArchiveHandler()
observer = Observer()
observer.schedule(handler, path='/data/incoming', recursive=False)
observer.start()

try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
observer.join()

二、自动归档(按日期 / 类型) ​

python
import shutil
from pathlib import Path
from datetime import datetime

def archive_files(src_dir, dest_dir):
    src = Path(src_dir)
    dest = Path(dest_dir) / datetime.now().strftime('%Y-%m-%d')
    dest.mkdir(parents=True, exist_ok=True)

    for file in src.iterdir():
        if file.is_file():
            target = dest / file.name
            shutil.move(str(file), str(target))
            print(f'移动 {file.name} → {dest}')

# 每天凌晨归档昨天的文件
archive_files('/data/incoming', '/data/archive')

三、批量重命名 ​

python
from pathlib import Path

def rename_files(directory, pattern, replacement, dry_run=True):
    """批量替换文件名中的字符"""
    for file in Path(directory).iterdir():
        if file.is_file() and pattern in file.name:
            new_name = file.name.replace(pattern, replacement)
            target = file.with_name(new_name)
            if dry_run:
                print(f'[预览] {file.name} → {new_name}')
            else:
                file.rename(target)
                print(f'[已重命名] {file.name} → {new_name}')

# 预览
rename_files('/data', ' ', '_', dry_run=True)
# 实际执行
rename_files('/data', ' ', '_', dry_run=False)

按日期前缀:

python
def add_date_prefix(directory, prefix_format='%Y%m%d_'):
    from datetime import datetime
    prefix = datetime.now().strftime(prefix_format)
    for file in Path(directory).iterdir():
        if file.is_file() and not file.name.startswith(prefix):
            new_name = prefix + file.name
            file.rename(file.with_name(new_name))

add_date_prefix('/data/exports')

四、PDF 合并 / 拆分 ​

python
import PyPDF2

def merge_pdfs(files, output):
    merger = PyPDF2.PdfMerger()
    for f in files:
        merger.append(f)
    merger.write(output)
    merger.close()

merge_pdfs(['a.pdf', 'b.pdf', 'c.pdf'], 'merged.pdf')

def split_pdf(input_file, output_prefix, pages_per_split=10):
    reader = PyPDF2.PdfReader(input_file)
    total = len(reader.pages)
    for i in range(0, total, pages_per_split):
        writer = PyPDF2.PdfWriter()
        for j in range(i, min(i + pages_per_split, total)):
            writer.add_page(reader.pages[j])
        out = f'{output_prefix}_{i // pages_per_split + 1}.pdf'
        with open(out, 'wb') as f:
            writer.write(f)

split_pdf('big.pdf', 'part')

五、图片处理(PIL) ​

python
from PIL import Image
import os

# 批量压缩
def compress_images(directory, quality=80):
    for file in Path(directory).glob('*.jpg'):
        img = Image.open(file)
        if img.size[0] > 1920:
            img.thumbnail((1920, 1920))
        img.save(file, quality=quality, optimize=True)
        print(f'压缩 {file.name}')

# 加水印
def add_watermark(image_path, text='内部资料'):
    img = Image.open(image_path).convert('RGBA')
    txt_layer = Image.new('RGBA', img.size, (255, 255, 255, 0))
    from PIL import ImageDraw, ImageFont
    draw = ImageDraw.Draw(txt_layer)
    font = ImageFont.truetype('msyh.ttf', 36)
    draw.text((50, 50), text, fill=(255, 255, 255, 128), font=font)
    result = Image.alpha_composite(img, txt_layer)
    result.save(image_path)

# 转格式
img = Image.open('a.png')
img.save('a.jpg', quality=90)

六、压缩 / 解压 ​

python
import zipfile, gzip, shutil

# ZIP
def zip_dir(src_dir, zip_path):
    with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zf:
        for root, dirs, files in os.walk(src_dir):
            for file in files:
                full = os.path.join(root, file)
                rel = os.path.relpath(full, src_dir)
                zf.write(full, rel)

def unzip(zip_path, dest_dir):
    with zipfile.ZipFile(zip_path, 'r') as zf:
        zf.extractall(dest_dir)

# tar.gz
import tarfile
def make_tar(src_dir, tar_path):
    with tarfile.open(tar_path, 'w:gz') as tf:
        tf.add(src_dir, arcname='.')

八、踩坑总结 ​

坑解决
文件被占用监听 .tmp 文件,完成后改名再处理
文件名中文乱码用 Path 而非字符串,设 encoding='utf-8'
大文件占用内存用 generator / 流式读取
路径不存在Path.mkdir(parents=True, exist_ok=True)
重命名失败(跨盘)用 shutil.move 而非 rename
处理失败无通知异常时发飞书 / 邮件 + 留现场
watchdog 性能异步事件 + 批量消费队列
