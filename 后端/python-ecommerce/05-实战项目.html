实战项目 ​

一个完整的跨境电商自动化系统 demo:商品采集 + 数据清洗 + 库存同步 + 报表推送。

项目目标 ​

从 1688 抓取指定供应商的商品数据
  ↓
清洗(去重、统一字段、价格标准化)
  ↓
推送到 Shopify
  ↓
每日定时同步库存
  ↓
生成日报推飞书

3. 1688 采集 ​

python
# src/collectors/alibaba.py
import undetected_playwright as up
from .base import BaseCollector

class AlibabaCollector(BaseCollector):
    def search(self, keyword, pages=3):
        with up.Chromium(headless=False) as browser:
            page = browser.new_page()
            results = []
            for pn in range(1, pages + 1):
                page.goto(f'https://s.1688.com/...&keywords={keyword}&beginPage={pn}')
                page.wait_for_load_state('networkidle')

                for item in page.query_selector_all('.sm-offer-item'):
                    try:
                        results.append(self.parse(item))
                    except Exception as e:
                        print(f'解析失败: {e}')
            return results

    def parse(self, item):
        return {
            'platform': '1688',
            'platform_id': item.get_attribute('data-offer-id'),
            'title': item.query_selector('.title').inner_text(),
            'price': float(item.query_selector('.price').inner_text().replace('¥', '')),
            'url': item.query_selector('a').get_attribute('href'),
        }

4. Shopify 采集 ​

python
# src/collectors/shopify.py
import requests
from .base import BaseCollector

class ShopifyCollector(BaseCollector):
    def search(self, keyword, pages=0):
        """Shopify 通常直接查中心库,这里留接口"""
        return self.search_db(keyword)

    def search_db(self, keyword):
        import sqlite3
        conn = sqlite3.connect(self.config.DB_PATH)
        rows = conn.execute(
            "SELECT * FROM products WHERE title LIKE ?",
            (f'%{keyword}%',)
        ).fetchall()
        return [dict(zip([c[0] for c in conn.execute('SELECT * FROM products').description], row)) for row in rows]

5. 数据清洗 ​

python
# src/cleaners/product.py
import pandas as pd
import re

class ProductCleaner:
    EXCHANGE_RATE = {'USD': 7.2, 'EUR': 7.8, 'CNY': 1.0}

    def clean(self, products):
        df = pd.DataFrame(products)

        # 1. 去重
        df = df.drop_duplicates(subset=['platform', 'platform_id'])

        # 2. 价格标准化为 CNY
        df['price_cny'] = df['price'] * df['currency'].map(self.EXCHANGE_RATE).fillna(1.0)

        # 3. 标题清洗
        df['title'] = df['title'].str.strip()
        df['title'] = df['title'].str.replace(r'\\s+', ' ', regex=True)

        # 4. 过滤无效
        df = df[df['price_cny'] > 0]
        df = df[df['title'].str.len() > 0]

        return df.to_dict('records')

6. 推送到 Shopify ​

python
# src/publishers/shopify.py
import requests

class ShopifyPublisher:
    def __init__(self, config):
        self.config = config

    def upsert_product(self, product):
        # Shopify 有 create / update,但没有 upsert
        existing = self.find_by_sku(product['platform_id'])
        if existing:
            return self.update(existing['id'], product)
        return self.create(product)

    def create(self, product):
        return requests.post(
            f'https://{self.config.SHOPIFY_DOMAIN}/admin/api/2024-01/products.json',
            headers={'X-Shopify-Access-Token': self.config.SHOPIFY_TOKEN},
            json={
                'product': {
                    'title': product['title'],
                    'variants': [{'sku': product['platform_id'], 'price': product['price_cny']}],
                }
            }
        ).json()

import requests

class FeishuPublisher:
    def send_report(self, title, summary):
        return requests.post(self.config.FEISHU_WEBHOOK, json={

8. 调度入口 ​

python
# scheduler.py
from apscheduler.schedulers.blocking import BlockingScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.triggers.interval import IntervalTrigger
from src.collectors.alibaba import AlibabaCollector
from src.cleaners.product import ProductCleaner
from src.publishers.shopify import ShopifyPublisher
from src.publishers.feishu import FeishuPublisher

config = Config()
collector = AlibabaCollector(config)
cleaner = ProductCleaner()
shopify_pub = ShopifyPublisher(config)
feishu_pub = FeishuPublisher(config)

def job_scrape_and_publish():
    # 1. 采集
    raw = collector.search('手机壳', pages=3)
    print(f'采集: {len(raw)} 条')

    # 2. 清洗
    clean = cleaner.clean(raw)
    print(f'清洗后: {len(clean)} 条')

    # 3. 推送 Shopify
    for product in clean:
        try:
            shopify_pub.upsert_product(product)
        except Exception as e:
            print(f'推送失败 {product["platform_id"]}: {e}')

    # 4. 飞书通知
    feishu_pub.send_report('商品同步完成', {
        '采集': len(raw),
        '清洗后': len(clean),
        '推送': len(clean),
    })

def job_sync_inventory():
    # 每 10 分钟同步库存
    from src.publishers.shopify import sync_inventory_to_shopify
    sync_inventory_to_shopify()

def job_daily_report():
    # 每天 9 点日报
    from src.reports.daily import generate_daily_report
    generate_daily_report(feishu_pub)

scheduler = BlockingScheduler(timezone='Asia/Shanghai')
scheduler.add_job(job_scrape_and_publish, IntervalTrigger(minutes=60), id='scrape')
scheduler.add_job(job_sync_inventory, IntervalTrigger(minutes=10), id='sync')
scheduler.add_job(job_daily_report, CronTrigger(hour=9, minute=0), id='daily')
scheduler.start()

# src/utils/monitor.py
from loguru import logger

def monitor_health():
    """健康检查:跑通核心链路 + 异常告警"""
    try:
        # 测采集
        raw = collector.search('test', pages=1)
        assert len(raw) > 0, '采集为空'

        # 测 Shopify 连接
        resp = requests.get(
            f'https://{config.SHOPIFY_DOMAIN}/admin/api/2024-01/shop.json',
            headers={'X-Shopify-Access-Token': config.SHOPIFY_TOKEN}
        )
        assert resp.status_code == 200, 'Shopify API 异常'

        logger.info('健康检查通过')
    except Exception as e:
        logger.exception(f'健康检查失败: {e}')
        feishu_pub.send_report('系统告警', {'错误': str(e)})

# 每 30 分钟跑一次健康检查
schedule.every(30).minutes.do(monitor_health)

总结 ​

这个项目覆盖了电商自动化的核心链路:

  • ✅ 数据采集(1688)
  • ✅ 数据清洗(pandas)
  • ✅ 多平台同步(Shopify)
  • ✅ 飞书告警 / 报表
  • ✅ 定时调度(APScheduler)
  • ✅ 健康监控

可以扩展:

  • 多采集源(Shopify / 淘宝 / 亚马逊)
  • 高级反爬(代理池、Cookie 池)
  • 复杂业务规则(限购、活动价)
  • 监控看板(Grafana)
