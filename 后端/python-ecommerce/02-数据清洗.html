数据清洗 ​

把采集来的脏数据洗干净:去重、字段映射、单位换算、数据校验。

一、基础数据清洗 ​

python
import pandas as pd
import re

df = pd.read_csv('raw_products.csv')

# 1. 去除重复行
df = df.drop_duplicates(subset=['product_id'])

# 2. 处理空值
df['price'] = df['price'].fillna(0)
df['title'] = df['title'].fillna('Untitled')

# 3. 字段清洗(去除空格、特殊字符)
df['title'] = df['title'].str.strip()
df['title'] = df['title'].str.replace(r'\\s+', ' ', regex=True)

# 4. 价格清洗(去掉货币符号、转数字)
def parse_price(s):
    if pd.isna(s):
        return 0.0
    if isinstance(s, (int, float)):
        return float(s)
    match = re.search(r'[\\d.]+', str(s))
    return float(match.group()) if match else 0.0

df['price_cny'] = df['price'].apply(parse_price)

# 5. 货币换算
EXCHANGE = {'USD': 7.2, 'EUR': 7.8, 'JPY': 0.05, 'CNY': 1.0}
df['price_cny'] = df.apply(
    lambda r: r['price'] * EXCHANGE.get(r['currency'], 1.0),
    axis=1
)

# 6. 单位换算(克、毫升、英寸)
def to_gram(value, unit):
    if unit == 'kg':
        return value * 1000
    elif unit == 'lb':
        return value * 453.592
    return value

df['weight_g'] = df.apply(lambda r: to_gram(r['weight'], r['unit']), axis=1)

二、字段映射(多平台统一) ​

不同平台字段名不同,需要映射到统一 schema:

python
PLATFORM_MAPPING = {
    '1688': {
        'productId': 'platform_id',
        'subject': 'title',
        'price': 'price',
    },
    'taobao': {
        'item_id': 'platform_id',
        'title': 'title',
        'price': 'price',
    }

def normalize(platform, raw_data):
    mapping = PLATFORM_MAPPING[platform]
    return {mapping.get(k, k): v for k, v in raw_data.items()}

# 使用
raw_1688 = {'productId': '123', 'subject': 'iPhone', 'price': 8999}
raw_taobao = {'item_id': '456', 'title': 'AirPods', 'price': 1299}

n1 = normalize('1688', raw_1688)  # {'platform_id': '123', 'title': 'iPhone', 'price': 8999}
n2 = normalize('taobao', raw_taobao)

三、数据校验(pydantic) ​

python
from pydantic import BaseModel, Field, validator
from typing import Optional

class Product(BaseModel):
    platform_id: str
    title: str = Field(..., min_length=1, max_length=500)
    price_cny: float = Field(..., ge=0)
    weight_g: Optional[float] = Field(None, ge=0)
    category: str

    @validator('title')
    def title_no_html(cls, v):
        if '<' in v and '>' in v:
            raise ValueError('Title contains HTML')
        return v.strip()

    @validator('price_cny')
    def price_reasonable(cls, v):
        if v > 1_000_000:
            raise ValueError('Price too high')
        return round(v, 2)

# 校验
try:
    p = Product(
        platform_id='123',
        title='iPhone 15',
        price_cny=8999,
        weight_g=300,
        category='手机'
    )
    print(p.json())
except ValidationError as e:
    print(e.json())

四、文本去重(相似度) ​

python
from difflib import SequenceMatcher

def similarity(a, b):
    return SequenceMatcher(None, a, b).ratio()

# 同一商品在两个平台标题略不同
t1 = 'Apple iPhone 15 Pro Max 256GB'
t2 = 'iPhone 15 Pro Max Apple 256G 全新'

sim = similarity(t1, t2)
print(f'相似度: {sim:.2f}')  # 0.65

# 大批量找重复
from itertools import combinations
items = df.to_dict('records')
seen = []
for i, j in combinations(range(len(items)), 2):
    sim = similarity(items[i]['title'], items[j]['title'])
    if sim > 0.85:
        seen.append((items[i]['platform_id'], items[j]['platform_id'], sim))

print(seen)

五、数据脱敏 ​

python
import re

def desensitize(text):
    """脱敏:手机号 / 邮箱 / 身份证"""
    # 手机号
    text = re.sub(r'1[3-9]\\d{9}', '1XX****XXXX', text)
    # 邮箱
    text = re.sub(r'([a-zA-Z0-9])[a-zA-Z0-9]+@', r'\\1***@', text)
    # 身份证
    text = re.sub(r'\\d{17}[\\dXx]', 'XXXXXXXXXXXXXXXXXXX', text)
    return text

print(desensitize('我的手机 13800138000,邮箱 alice@example.com'))
# '我的手机 1XX****XXXX,邮箱 a***@example.com'

六、批量入库前检查 ​

python
def clean_dataframe(df):
    """数据清洗流水线"""
    initial_count = len(df)

    # 1. 去重
    df = df.drop_duplicates(subset=['platform_id'])

    # 2. 去除无效数据
    df = df[df['title'].str.len() > 0]
    df = df[df['price_cny'] > 0]

    # 3. 字段清洗
    df['title'] = df['title'].str.strip()
    df['price_cny'] = df['price_cny'].round(2)

    # 4. 校验
    df = df[df['title'].str.match(r'^[\u4e00-\u9fa5a-zA-Z0-9\s\-]+$')]

    final_count = len(df)
    print(f'清洗前: {initial_count},清洗后: {final_count},丢弃率: {(initial_count - final_count) / initial_count * 100:.1f}%')
    return df

df = clean_dataframe(df)

七、踩坑总结 ​

坑解决
同一商品不同标题用 difflib / 嵌入向量计算相似度
多平台字段名不同建映射表,统一 schema
价格含税/汇率混乱单独字段存原价 + 币种 + 标准化字段
数据格式不规范用 pydantic 校验,日志记录异常
编码问题(GBK vs UTF-8)pd.read_csv(encoding='utf-8-sig')
大文件清洗慢dask / polars 替代 pandas
