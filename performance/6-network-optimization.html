网络与资源优化详解 ​

一、资源加载策略 ​

1.1 preload vs prefetch ​

html
<!-- preload：预加载当前页面必需资源 -->
<!-- 浏览器会在页面加载时立即加载资源 -->
<link rel="preload" href="main.js" as="script">
<link rel="preload" href="font.woff2" as="font" crossorigin>
<link rel="preload" href="hero.webp" as="image">

<!-- prefetch：预取未来可能使用的资源 -->
<!-- 浏览器在空闲时加载资源 -->
<link rel="prefetch" href="next-page.js" as="script">
<link rel="prefetch" href="next-page.css" as="style">
特性preloadprefetch
加载时机立即空闲时
优先级高低
缓存与主资源一致可复用缓存
用途当前页面必需未来可能用

1.2 脚本加载策略 ​

html
<!-- 无属性：立即执行，阻塞 HTML 解析 -->
<script src="app.js"></script>

<!-- defer：HTML 解析完成后执行，不阻塞 -->
<script defer src="app.js"></script>

<!-- async：下载完成后立即执行，不阻塞（可能阻塞 load）-->
<script async src="analytics.js"></script>
html
<!-- 加载顺序对比 -->

<!-- defer（按文档顺序） -->
<script defer src="a.js"></script>
<script defer src="b.js"></script>
<!-- 执行顺序：a.js → b.js -->

<!-- async（谁先下载完谁执行） -->
<script async src="a.js"></script>
<script async src="b.js"></script>
<!-- 执行顺序：不确定

1.3 CSS 加载策略 ​

html
<!-- 阻塞渲染 -->
<link rel="stylesheet" href="critical.css">

<!-- 媒体查询：不阻塞（仅在匹配时阻塞）-->
<link rel="stylesheet" href="print.css" media="print">

<!-- 异步加载 CSS -->
<link rel="stylesheet" href="non-critical.css" media="print" onload="this.media='all'">

<!-- 或使用 preload + onload -->
<link rel="preload" href="styles.css" as="style" onload="this.onload=null;this.rel='stylesheet'">

2.3 缓存策略配置 ​

javascript
// Express 设置缓存
app.use('/static', express.static('public', {
  maxAge: '1y',           // 1年
  etag: true,             // 支持 ETag

  setHeaders: (res, path) => {

}

# HTML 不缓存
location ~* \.html$ {
  expires -1;
  add_header Cache-Control "no-store, no-cache, must-revalidate";
}

javascript
// JS 懒加载（兼容不支持原生懒加载的浏览器）
const observer = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target
      img.src = img.dataset.src
      observer.unobserve(img)
    }
  })
})

document.querySelectorAll('img[data-src]').forEach(img => {
  observer.observe(img)
})

<!-- base64 占位（内联，小图）-->

<!-- blur 占位（模糊图）-->

3.5 CSS Sprites vs Icon Font vs SVG ​

方案优点缺点适用场景
CSS SpritesHTTP 请求少维护困难，不能缩放图标固定
Icon Font矢量，可缩放渲染复杂，只支持单色简单图标
SVG矢量，多色，语义化旧浏览器兼容现代应用
Base64无额外请求增大 CSS 体积极小图标
html
<!-- SVG 内联 -->
<svg width="24" height="24" viewBox="0 0 24 24">
  <path d="M..."/>
</svg>

<!-- SVG Sprite -->
<svg>
  <use xlink:href="#icon-name"/>
</svg>

4.3 域名分散 vs 收敛 ​

策略优点缺点
域名收敛减少 DNS 查询，复用连接并发受限

// 分散：静态资源用不同域名

static2.example.com
static3.example.com

5.1 font-display ​

css
@font-face {

  src: url('font.woff2') format('woff2');

七、请求优化 ​

7.1 防抖节流 ​

javascript
// 防抖：事件触发 n 秒后执行，n 秒内再次触发则重置
function debounce(fn, delay) {
  let timer = null
  return function(...args) {
    clearTimeout(timer)
    timer = setTimeout(() => {
      fn.apply(this, args)
    }, delay)
  }
}

// 节流：n 秒内只执行一次
function throttle(fn, delay) {
  let last = 0
  return function(...args) {
    const now = Date.now()
    if (now - last >= delay) {
      last = now
      fn.apply(this, args)
    }
  }
}

// 使用
const handleSearch = debounce((query) => {
  fetch(`/api/search?q=${query}`)
}, 300)

input.addEventListener('input', (e) => handleSearch(e.target.value))

7.2 请求合并 ​

javascript
// 请求队列，批量发送
class RequestQueue {
  constructor() {
    this.queue = []
    this.timer = null
  }

  add(request) {
    this.queue.push(request)
    this.scheduleFlush()
  }

  scheduleFlush() {
    if (this.timer) return
    this.timer = setTimeout(() => {
      this.flush()
    }, 100) // 批量发送
  }

  flush() {
    const requests = this.queue.splice(0)
    requests.forEach(({ resolve, reject, config }) => {
      axios(config).then(resolve).catch(reject)
    })
    this.timer = null
  }
}

八、Service Worker ​

8.1 基础缓存 ​

javascript
// sw.js
const CACHE_NAME = 'v1'
const urlsToCache = [
  '/',
  '/static/js/main.js',
  '/static/css/main.css'
]

// 安装
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(urlsToCache))
  )
})

// 拦截请求
self.addEventListener('fetch', (event) => {
  event.respondWith(
    caches.match(event.request)
      .then(response => {
        // 缓存命中返回缓存，否则网络请求
        return response || fetch(event.request)
      })
  )
})

8.2 缓存策略 ​

策略说明适用场景
Cache First先缓存，失败再网络静态资源
Network First先网络，失败再缓存动态数据
Stale While Revalidate返回缓存，后台更新经常变化的资源
Network Only只用网络实时数据
javascript
// Stale While Revalidate
self.addEventListener('fetch', (event) => {
  event.respondWith(
    caches.open(CACHE_NAME).then(cache => {
      return cache.match(event.request).then(response => {
        const fetchPromise = fetch(event.request).then(networkResponse => {
          cache.put(event.request, networkResponse.clone())
          return networkResponse
        })
        return response || fetchPromise
      })
    })
  )
})

🎯 面试常考 ​

Q1：静态资源应该用什么缓存策略?

带内容哈希的文件名(fingerprint)用 Cache-Control: public, max-age=31536000, immutable 长缓存;HTML 入口不缓存或 no-cache,保证发版后能拿到新版本。

Q2：preload 和 prefetch 有什么区别?

preload 是「现在就要用,提前拉」,用于关键资源;prefetch 是「可能要用,空闲时拉」,用于低优先级资源。滥用 preload 会抢占带宽反而变慢。

📝 小结 ​

  • 网络优化的四个抓手:缓存头策略、连接复用(HTTP/2 多路复用 + keep-alive)、请求合并与防抖、资源体积(图片格式 / 字体子集化)。
  • preload / prefetch / preconnect 各管一段链路,用错比不用更糟。
  • 图片是最容易见效的一环:AVIF/WebP + 响应式尺寸 + 懒加载 + 显式宽高,四项一起做。
  • 字体用 font-display: swap + 子集化,避免 FOIT 阻塞首屏文本渲染。
